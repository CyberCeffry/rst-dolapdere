  Future<void> _syncCollection<T>({
    required String name,
    required List<T> Function() local,
    required T Function(Map<String, dynamic>) decode,
    required void Function(List<T>) assign,
  }) async {
    final collection = _collection(name);
    final tombstones = _collection('_sync_deletions');
    final lastSync = lastSuccessfulSync;
    final pending = _readPendingChanges();
    final collectionPending = pending
        .where((item) => item['collection'] == name)
        .toList();

    final remaining = pending
        .where((item) => item['collection'] != name)
        .toList();

    final values = <String, T>{
      for (final item in local()) (item as dynamic).id as String: item,
    };

    DateTime readTime(dynamic value) {
      return DateTime.tryParse(value?.toString() ?? '') ??
          DateTime.fromMillisecondsSinceEpoch(0, isUtc: true);
    }

    for (final change in collectionPending) {
      final id = change['id'] as String;
      final localTime = readTime(change['updatedAt']);
      final remote = await collection.doc(id).get();
      final deletedRef = tombstones.doc('${name}_$id');
      final remoteDeleted = await deletedRef.get();

      final remoteTime = readTime(remote.data()?['_syncUpdatedAt']);
      final deletedTime = readTime(remoteDeleted.data()?['_syncUpdatedAt']);

      if (remoteTime.isAfter(localTime) || deletedTime.isAfter(localTime)) {
        if (deletedTime.isAfter(remoteTime)) {
          values.remove(id);
        } else if (remote.exists) {
          values[id] = decode({...remote.data()!, 'id': id});
        }
        continue;
      }

      if (change['deleted'] == true) {
        await deletedRef.set({
          'collection': name,
          'recordId': id,
          '_syncUpdatedAt': localTime.toIso8601String(),
        });
        await collection.doc(id).delete();
        values.remove(id);
      } else {
        final data = Map<String, dynamic>.from(change['data'] as Map);
        await collection.doc(id).set({
          ...data,
          '_syncUpdatedAt': localTime.toIso8601String(),
        });
        await deletedRef.delete();
      }
    }

    final changedDocs = lastSync == null
        ? await collection.get()
        : await collection
              .where(
                '_syncUpdatedAt',
                isGreaterThan: lastSync.toUtc().toIso8601String(),
              )
              .get();

    for (final document in changedDocs.docs) {
      final id = document.id;
      final deleted = await tombstones.doc('${name}_$id').get();
      final documentTime = readTime(document.data()['_syncUpdatedAt']);
      final deletedTime = readTime(deleted.data()?['_syncUpdatedAt']);

      if (deletedTime.isAfter(documentTime) ||
          deletedTime.isAtSameMomentAs(documentTime)) {
        values.remove(id);
      } else {
        values[id] = decode({...document.data(), 'id': id});
      }
    }

    final changedDeletes = lastSync == null
        ? await tombstones.where('collection', isEqualTo: name).get()
        : await tombstones
              .where('collection', isEqualTo: name)
              .where(
                '_syncUpdatedAt',
                isGreaterThan: lastSync.toUtc().toIso8601String(),
              )
              .get();

    for (final document in changedDeletes.docs) {
      values.remove(document.data()['recordId'] as String);
    }

    assign(values.values.toList());
    await _saveList(name, values.values.toList());
    await _savePendingChanges([
      ...remaining,
      ...collectionPending.where((change) {
        final id = change['id'] as String;
        return false;
      }),
    ]);
  }

  Future<void> _mergeCollection<T>({
    required String name,
    required List<T> Function() local,
    required T Function(Map<String, dynamic>) decode,
    required String Function(T) idOf,
    required void Function(List<T>) assign,
  }) async {
    final collection = _collection(name);
    final snapshot = await _queryCollection(name).get();

    final merged = <String, T>{
      if (_user?.role != AppRole.supervisor)
        for (final item in local()) idOf(item): item,
    };

    final remoteIds = <String>{};

    for (final document in snapshot.docs) {
      merged[document.id] = decode({...document.data(), 'id': document.id});

      remoteIds.add(document.id);
    }

    final values = merged.values.toList();

    assign(values);
    await _saveList(name, values);

    if (_user?.role == AppRole.supervisor) return;
    final missing = merged.entries.where(
      (entry) => !remoteIds.contains(entry.key),
    );

    for (final entry in missing) {
      unawaited(
        _writeDocument(
          collection.doc(entry.key),
          (entry.value as dynamic).toJson(),
        ),
      );
    }
  }

  void _listenCollection<T>(
    String name,
    T Function(Map<String, dynamic>) decode,
    void Function(List<T>) assign,
  ) {
    _subscriptions.add(
      _queryCollection(name).snapshots().listen((snapshot) {
        if (snapshot.metadata.isFromCache) return;

        try {
          final values = snapshot.docs
              .map(
                (document) => decode({...document.data(), 'id': document.id}),
              )
              .toList();

          assign(values);

          unawaited(_saveList(name, values));

          if (name == 'contracts') {
            unawaited(_refreshAvailability());
          }

          syncError = null;
          notifyListeners();
        } catch (error) {
          _setSyncError(error);
        }
      }, onError: _setSyncError),
    );
  }

  Query<Map<String, dynamic>> _queryCollection(String name) {

      return collection.where('responsibleUserId', isEqualTo: _user!.uid);
    }
    final ids = name == 'customers'
        ? _user!.assignedCustomerIds
        : name == 'adSpaces'
        ? _user!.assignedAdSpaceIds
        : const <String>[];
    if (ids.isEmpty) {
      return collection.where(
        FieldPath.documentId,
        isEqualTo: '__no_assigned_records__',
      );
    }
    return collection.where(
      FieldPath.documentId,
      whereIn: ids.take(30).toList(),
    );
  }

  Future<void> _writeDocument(
    DocumentReference<Map<String, dynamic>> reference,
    Map<String, dynamic> data,
  ) => _queue(() => reference.set(data));

  Future<void> _writeCollection(String name, List<dynamic> values) async {
    final collection = _collection(name);

    for (var start = 0; start < values.length; start += 450) {
      final batch = _firestore!.batch();

      for (final value in values.skip(start).take(450)) {
        batch.set(
          collection.doc(value.id as String),
          value.toJson() as Map<String, dynamic>,
        );
      }

      await batch.commit();
    }
  }

  Future<void> _queue(Future<void> Function() operation) {
    final next = _writeQueue.then((_) async {
      if (_firestore == null) return;

      _isSyncing = true;
      notifyListeners();

      try {
        await operation();
        syncError = null;
      } catch (error) {
        _setSyncError(error);
      } finally {
        _isSyncing = false;
        notifyListeners();
      }
    });

    _writeQueue = next.catchError((_) {});

    return next;
  }

  Future<void> _persistCollection(String name, List<dynamic> values) async {
    final currentValues = name == 'customers'
        ? _customers
        : name == 'adSpaces'
        ? _adSpaces
        : _contracts;

    await _saveList(name, currentValues);

    for (final value in values) {
      await _addPendingChange(
        collection: name,
        id: value.id as String,
        deleted: false,
        data: value.toJson() as Map<String, dynamic>,
      );
    }
  }

  Future<void> _deleteRemote(String name, String id) async {
    await _addPendingChange(collection: name, id: id, deleted: true);
  }

  Map<String, dynamic> get _settingsJson => {

    'reminderOffsets': reminderOffsets,
    'workingDays': workingDays,

  Future<void> _applySettings(Map<String, dynamic> values) async {
    phoneRemindersEnabled =
        values['phoneRemindersEnabled'] as bool? ?? phoneRemindersEnabled;

    windowsRemindersEnabled =
        values['windowsRemindersEnabled'] as bool? ?? windowsRemindersEnabled;

    reminderTimeMinutes =
        (values['reminderTimeMinutes'] as num?)?.toInt() ?? reminderTimeMinutes;
    final configuredOffsets = values['reminderOffsets'];
    if (configuredOffsets is List) {
      final parsedOffsets =
          configuredOffsets
              .whereType<num>()
              .map((value) => value.toInt())
              .where(availableReminderOffsets.contains)
              .toSet()
              .toList()
            ..sort((a, b) => b.compareTo(a));
      if (parsedOffsets.isNotEmpty) reminderOffsets = parsedOffsets;
    }

    final configuredWorkingDays = values['workingDays'];
    if (configuredWorkingDays is List) {
      final parsedDays =
          configuredWorkingDays
              .whereType<num>()
              .map((value) => value.toInt())
              .where(weekdayLabels.containsKey)
              .toSet()
              .toList()
            ..sort();
      if (parsedDays.isNotEmpty) workingDays = parsedDays;
    }

    await _preferences.setBool('phoneRemindersEnabled', phoneRemindersEnabled);
    await _preferences.setBool(
      'windowsRemindersEnabled',
      windowsRemindersEnabled,
    );

    await _preferences.setInt('reminderTimeMinutes', reminderTimeMinutes);
    await _preferences.setStringList(
      'reminderOffsets',
      reminderOffsets.map((offset) => offset.toString()).toList(),
    );
    await _preferences.setStringList(
      'workingDays',
      workingDays.map((weekday) => weekday.toString()).toList(),
    );
    notifyListeners();
  }

  Future<void> saveContract(ContractModel contract) async {
    if (_user?.role == AppRole.supervisor) {
      await _authService.submitContractChange(
        operation: 'create',
        contract: contract,
      );
      return;
    }
    final existing = _contracts.where((item) => item.id == contract.id);
    final existingContract = existing.firstOrNull;

    if (existing.isNotEmpty &&
        (existing.first.isFinalized ||
            DateHelper.daysUntil(existing.first.endDate) < 0)) {
      throw StateError(
        'Biten sözleşme yeniden açılamaz. Devam etmek için Uzat seçeneğini kullanın.',
      );
    }

    final overlaps = _contracts.any(
      (item) =>
          item.id != contract.id &&
          item.adSpaceId == contract.adSpaceId &&
          item.isActive &&
          !item.isFinalized &&
          !DateHelper.daysUntil(
            item.endDate,
            from: contract.startDate,
          ).isNegative &&
          !DateHelper.daysUntil(
            contract.endDate,
            from: item.startDate,
          ).isNegative,
    );

    if (contract.isActive && overlaps) {
      throw StateError('Bu reklam alanında devam eden başka bir sözleşme var.');
    }

    final savedContract = ContractModel(
      id: contract.id,
      customerId: contract.customerId,
      adSpaceId: contract.adSpaceId,
      startDate: contract.startDate,
      endDate: contract.endDate,
      monthlyPrice: contract.monthlyPrice,
      durationUnit: contract.durationUnit,
      durationCount: contract.durationCount,
      notes: contract.notes,
      isActive: contract.isActive,
      isFinalized: contract.isFinalized,
      finalizedAt: contract.finalizedAt,
      renewedFromContractId: contract.renewedFromContractId,
      responsibleUserId:
          contract.responsibleUserId ??
          existingContract?.responsibleUserId ??
          '',
      responsibleUsername:
          contract.responsibleUsername ?? existingContract?.responsibleUsername,
      history:
          existingContract?.history ??
          [
            ContractHistoryEntry(
              action: ContractHistoryAction.created,
              occurredAt: DateTime.now(),
              newEndDate: contract.endDate,
              newMonthlyPrice: contract.monthlyPrice,
              actorUid: _user?.uid,
              actorUsername: _user?.username,
            ),
          ],
    );

    _contracts = _upsert(_contracts, savedContract, (item) => item.id);

    await _persistCollection('contracts', [savedContract]);
    await _refreshAvailability();

    notifyListeners();
  }

  Future<void> markContractEnded(String id) async {
    final index = _contracts.indexWhere((item) => item.id == id);

    if (index < 0) {
      throw StateError('Sözleşme bulunamadı.');
    }

    final current = _contracts[index];

    if (_user?.role == AppRole.supervisor) {
      await _authService.setContractActive(contractId: id, isActive: false);
      await setContractActive(id, false, alreadyUpdatedRemotely: true);
      return;
    }

    if (current.isFinalized) return;

    final finalized = _copyContract(
      current,
      isActive: false,
      isFinalized: true,
      finalizedAt: DateTime.now(),
      history: [
        ...current.history,
        ContractHistoryEntry(
          action: ContractHistoryAction.ended,
          occurredAt: DateTime.now(),
          previousEndDate: current.endDate,
          newEndDate: current.endDate,
          previousMonthlyPrice: current.monthlyPrice,
          newMonthlyPrice: current.monthlyPrice,
          actorUid: _user?.uid,
          actorUsername: _user?.username,
        ),
      ],
    );

    _contracts[index] = finalized;

    await _persistCollection('contracts', [finalized]);
    await _refreshAvailability();

    notifyListeners();
  }

      );
    }

      id: source.id,
      customerId: source.customerId,
      adSpaceId: source.adSpaceId,
      startDate: source.startDate,
      endDate: renewed.endDate,
      monthlyPrice: renewed.monthlyPrice,
      durationUnit: renewed.durationUnit,
      durationCount: renewed.durationCount,
      notes: renewed.notes,
      isActive: true,
      isFinalized: false,
      finalizedAt: null,
      renewedFromContractId: source.renewedFromContractId,
      responsibleUserId: source.responsibleUserId,
      responsibleUsername: source.responsibleUsername,

  Future<void> saveReminderSettings({
    required bool phone,
    bool? windows,
    int? reminderMinutes,
    List<int>? reminderOffsets,
    List<int>? workingDays,
  }) async {
    List<int>? selectedOffsets;
    if (reminderOffsets != null) {
      selectedOffsets =
          reminderOffsets
              .where(availableReminderOffsets.contains)
              .toSet()
              .toList()
            ..sort((a, b) => b.compareTo(a));
      if (selectedOffsets.isEmpty) {
        throw ArgumentError.value(
          reminderOffsets,
          'reminderOffsets',
          'At least one valid reminder day must be selected.',
        );
      }
    }
    List<int>? selectedWorkingDays;
    if (workingDays != null) {
      selectedWorkingDays =
          workingDays.where(weekdayLabels.containsKey).toSet().toList()..sort();
      if (selectedWorkingDays.isEmpty) {
        throw ArgumentError.value(
          workingDays,
          'workingDays',
          'At least one valid weekday must be selected.',
        );
      }
    }

    phoneRemindersEnabled = phone;
    windowsRemindersEnabled = windows ?? windowsRemindersEnabled;

    reminderTimeMinutes = reminderMinutes ?? reminderTimeMinutes;
    if (selectedOffsets != null) this.reminderOffsets = selectedOffsets;
    if (selectedWorkingDays != null) this.workingDays = selectedWorkingDays;

    await _preferences.setBool('phoneRemindersEnabled', phone);
    await _preferences.setBool(
      'windowsRemindersEnabled',
      windowsRemindersEnabled,
    );

    await _preferences.setInt('reminderTimeMinutes', reminderTimeMinutes);
    await _preferences.setStringList(
      'reminderOffsets',
      this.reminderOffsets.map((offset) => offset.toString()).toList(),
    );
    await _preferences.setStringList(
      'workingDays',
      this.workingDays.map((weekday) => weekday.toString()).toList(),
    );

    unawaited(_writeCloudSettings());

    notifyListeners();
  }

  Future<void> setContractActive(
    String id,
    bool isActive, {
    bool alreadyUpdatedRemotely = false,
  }) async {
    final index = _contracts.indexWhere((item) => item.id == id);
    if (index < 0) throw StateError('Sözleşme bulunamadı.');
    if (_user?.role == AppRole.supervisor && !alreadyUpdatedRemotely) {
      await _authService.setContractActive(contractId: id, isActive: isActive);
    }
    final source = _contracts[index];
    final now = DateTime.now();
    final updated = _copyContract(
      source,
      isActive: isActive,
      isFinalized: !isActive,
      finalizedAt: isActive ? null : now,
      history: [
        ...source.history,
        ContractHistoryEntry(
          action: isActive
              ? ContractHistoryAction.activated
              : ContractHistoryAction.ended,
          occurredAt: now,
          previousEndDate: source.endDate,
          newEndDate: source.endDate,
          previousMonthlyPrice: source.monthlyPrice,
          newMonthlyPrice: source.monthlyPrice,
          actorUid: _user?.uid,
          actorUsername: _user?.username,
        ),
      ],
    );
    _contracts[index] = updated;
    if (_user?.role == AppRole.supervisor) {
      await _saveList('contracts', _contracts);
    } else {
      await _persistCollection('contracts', [updated]);
    }
    await _refreshAvailability();
    notifyListeners();
  }

  List<ContractModel> get activeContracts {
    final today = DateTime.now();

    return _contracts.where((item) {
      return item.isActive &&
          !item.isFinalized &&
          DateHelper.isOnOrBeforeToday(item.startDate, today: today) &&
          DateHelper.isOnOrAfterToday(item.endDate, today: today);
    }).toList();
  }

  List<ContractModel> get endedContracts => _contracts.where((item) {
    return item.isFinalized || DateHelper.daysUntil(item.endDate) < 0;
  }).toList()..sort((a, b) => b.endDate.compareTo(a.endDate));

  List<ContractModel> get expiringContracts => activeContracts.where((item) {
    final days = DateHelper.daysUntil(item.endDate);
    return days >= 0 && days <= 10;
  }).toList()..sort((a, b) => a.endDate.compareTo(b.endDate));

  List<T> _upsert<T>(List<T> items, T value, String Function(T) idOf) {
    final result = List<T>.of(items);

    final index = result.indexWhere((item) => idOf(item) == idOf(value));

    if (index < 0) {
      result.add(value);
    } else {
      result[index] = value;
    }

    return result;
  }

  ContractModel _copyContract(
    ContractModel source, {
    bool? isActive,
    bool? isFinalized,
    DateTime? finalizedAt,
    String? renewedFromContractId,
    List<ContractHistoryEntry>? history,
    String? responsibleUserId,
    String? responsibleUsername,
  }) {
    return ContractModel(
      id: source.id,
      customerId: source.customerId,
      adSpaceId: source.adSpaceId,
      startDate: source.startDate,
      endDate: source.endDate,
      monthlyPrice: source.monthlyPrice,
      durationUnit: source.durationUnit,
      durationCount: source.durationCount,
      notes: source.notes,
      isActive: isActive ?? source.isActive,
      isFinalized: isFinalized ?? source.isFinalized,
      finalizedAt: finalizedAt ?? source.finalizedAt,
      renewedFromContractId:
          renewedFromContractId ?? source.renewedFromContractId,
      history: history ?? source.history,
      responsibleUserId: responsibleUserId ?? source.responsibleUserId,
      responsibleUsername: responsibleUsername ?? source.responsibleUsername,
    );
  }

  Future<void> _refreshAvailability() async {
    final now = DateTime.now();

    _adSpaces = _adSpaces.map((space) {
      final occupied = _contracts.any(
        (contract) =>
            contract.adSpaceId == space.id &&
            contract.isActive &&
            !contract.isFinalized &&
            DateHelper.isOnOrBeforeToday(contract.startDate, today: now) &&
            DateHelper.isOnOrAfterToday(contract.endDate, today: now),
      );

      return AdSpaceModel(
        id: space.id,
        name: space.name,
        location: space.location,
        description: space.description,
        monthlyPrice: space.monthlyPrice,
        isAvailable: !occupied,
      );
    }).toList();

    await _saveList('adSpaces', _adSpaces);
  }

  Future<AppUser> signIn({
    required String username,
    required String password,
  }) async {
    final normalized = username.trim().toLowerCase();
    if (normalized.isEmpty || password.isEmpty) {
      throw ArgumentError('Kullanıcı adı ve şifre zorunludur.');
    }
    final credential = await _auth.signInWithEmailAndPassword(
      email: '$normalized@$_usernameDomain',
      password: password,
    );
    final user = credential.user;
    if (user == null) throw StateError('Oturum açılamadı.');
    final profile = await loadProfile(user.uid);
    if (profile == null || !profile.isActive) {
      await _auth.signOut();
      throw StateError('Kullanıcı hesabı etkin değil. Yöneticiyle görüşün.');
    }
    return profile;
  }

  Future<void> changePassword({
    required String currentPassword,
    required String newPassword,
  }) async {
    final user = _auth.currentUser;
    final email = user?.email;
    if (user == null || email == null) throw StateError('Oturum bulunamadı.');
    final credential = EmailAuthProvider.credential(
      email: email,
      password: currentPassword,
    );
    await user.reauthenticateWithCredential(credential);
    await user.updatePassword(newPassword);
  }

  Future<void> createUser({
    required String username,
    required String initialPassword,
    required AppRole role,
    required bool isActive,
    required Map<String, bool> permissions,
    String email = '',
  }) async {
    final normalizedUsername = username.trim().toLowerCase();
    if (!RegExp(r'^[a-z0-9._-]{3,32}$').hasMatch(normalizedUsername)) {
      throw ArgumentError('Kullanıcı adı 3-32 karakter olmalı.');
    }
    if (initialPassword.length < 10) {
      throw ArgumentError('Şifre en az 10 karakter olmalı.');
    }
    final indexReference = _firestore
        .collection('workspaces')
        .doc(workspaceId)
        .collection('usernameIndex')
        .doc(normalizedUsername);
    if ((await indexReference.get()).exists) {
      throw FirebaseException(
        plugin: 'cloud_firestore',
        code: 'already-exists',
        message: 'Bu kullanıcı adı zaten kullanımda.',
      );
    }
    final authApp = await Firebase.initializeApp(
      name: 'user-provision-${DateTime.now().microsecondsSinceEpoch}',
      options: Firebase.app().options,
    );
    final provisioningAuth = FirebaseAuth.instanceFor(app: authApp);
    String? createdUid;
    try {
      final credential = await provisioningAuth.createUserWithEmailAndPassword(
        email: '$normalizedUsername@$_usernameDomain',
        password: initialPassword,
      );
      final uid = credential.user!.uid;
      createdUid = uid;
      final userReference = _users.doc(uid);
      final batch = _firestore.batch();
      batch.set(indexReference, {'uid': uid});
      batch.set(userReference, {
        'username': normalizedUsername,
        'authEmail': '$normalizedUsername@$_usernameDomain',
        'email': email.trim(),
        'role': role.name,
        'isActive': isActive,
        'permissions': permissions,
        'notificationPreferences': {'inApp': true, 'push': true},
        'assignedContractIds': <String>[],
        'assignedCustomerIds': <String>[],
        'assignedAdSpaceIds': <String>[],
        'createdAt': FieldValue.serverTimestamp(),
      });
      await batch.commit();
    } catch (error) {
      if (createdUid != null) {
        await provisioningAuth.currentUser?.delete().catchError((_) {});
      }
      if (error is FirebaseException && error.code == 'permission-denied') {
        final racedIndex = await indexReference.get();
        if (racedIndex.exists) {
          throw FirebaseException(
            plugin: 'cloud_firestore',
            code: 'already-exists',
            message: 'Bu kullanıcı adı başka bir yönetici tarafından alındı.',
          );
        }
      }
      rethrow;
    } finally {
      await provisioningAuth.signOut();
      await authApp.delete();
    }
  }

  Future<void> assignContract({
    required String contractId,
    required String responsibleUserId,
  }) async {
    final actorUid = _auth.currentUser?.uid;
    if (actorUid == null) throw StateError('Oturum bulunamadı.');
    final contractReference = _firestore
        .collection('workspaces')
        .doc(workspaceId)
        .collection('contracts')
        .doc(contractId);
    final actorReference = _users.doc(actorUid);
    final targetReference = _users.doc(responsibleUserId);
    final initialContractSnapshot = await contractReference.get();
    final initialOldResponsibleId =
        initialContractSnapshot.data()?['responsibleUserId'] as String?;
    var previousContracts = <QueryDocumentSnapshot<Map<String, dynamic>>>[];
    if (initialOldResponsibleId != null &&
        initialOldResponsibleId.isNotEmpty &&
        initialOldResponsibleId != responsibleUserId) {
      final previousContractsQuery = await _firestore
          .collection('workspaces')
          .doc(workspaceId)
          .collection('contracts')
          .where('responsibleUserId', isEqualTo: initialOldResponsibleId)
          .get();
      previousContracts = previousContractsQuery.docs;
    }
    await _firestore.runTransaction((transaction) async {
      final contractSnapshot = await transaction.get(contractReference);
      final actorSnapshot = await transaction.get(actorReference);
      final targetSnapshot = responsibleUserId == actorUid
          ? actorSnapshot
          : await transaction.get(targetReference);
      if (!contractSnapshot.exists || !targetSnapshot.exists) {
        throw StateError('Sözleşme veya kullanıcı bulunamadı.');
      }
      final contract = contractSnapshot.data()!;
      final actor = actorSnapshot.data()!;
      final target = targetSnapshot.data()!;
      final role = actor['role'] as String?;
      final oldResponsibleId = contract['responsibleUserId'] as String?;
      if (oldResponsibleId != initialOldResponsibleId) {
        throw StateError('Sorumlu aynı anda değiştirildi. Yeniden deneyin.');
      }
      if (target['isActive'] != true) {
        throw StateError('Pasif kullanıcı sorumlu atanamaz.');
      }
      if (role == AppRole.supervisor.name) {
        if (responsibleUserId != actorUid) {
          throw StateError('Sorumlu yalnızca kendisini atayabilir.');
        }
        if (oldResponsibleId != null &&
            oldResponsibleId.isNotEmpty &&
            oldResponsibleId != actorUid) {
          throw StateError(
            'Başka bir sorumluya atanmış sözleşmeyi devralamazsınız.',
          );
        }
      } else if (role != AppRole.admin.name && role != AppRole.manager.name) {
        throw StateError('Sorumlu atama yetkiniz yok.');
      }

      DocumentSnapshot<Map<String, dynamic>>? previousUserSnapshot;
      if (oldResponsibleId != null &&
          oldResponsibleId.isNotEmpty &&
          oldResponsibleId != responsibleUserId) {
        final previousReference = _users.doc(oldResponsibleId);
        previousUserSnapshot = await transaction.get(previousReference);
      }

      final now = DateTime.now();
      transaction.update(contractReference, {
        'responsibleUserId': responsibleUserId,
        'responsibleUsername': target['username'] ?? '',
        'updatedAt': FieldValue.serverTimestamp(),
        'history': FieldValue.arrayUnion([
          {
            'action': 'assigned',
            'occurredAt': now.toIso8601String(),
            'actorUid': actorUid,
            'actorUsername': actor['username'] ?? '',
            'targetUserId': responsibleUserId,
            'targetUsername': target['username'] ?? '',
          },
        ]),
      });
      transaction.update(targetReference, {
        'assignedContractIds': FieldValue.arrayUnion([contractId]),
        'assignedCustomerIds': FieldValue.arrayUnion([
          contract['customerId'] as String,
        ]),
        'assignedAdSpaceIds': FieldValue.arrayUnion([
          contract['adSpaceId'] as String,
        ]),
        if (role == AppRole.supervisor.name) ...{
          'lastAssignedContractId': contractId,
          'lastAssignedCustomerId': contract['customerId'],
          'lastAssignedAdSpaceId': contract['adSpaceId'],
        },
      });
      if (previousUserSnapshot?.exists == true) {
        final previousProfile = previousUserSnapshot!.data()!;
        final remaining = previousContracts.where(
          (document) => document.id != contractId,
        );
        final hasCustomer = remaining.any(
          (document) => document.data()['customerId'] == contract['customerId'],
        );
        final hasAdSpace = remaining.any(
          (document) => document.data()['adSpaceId'] == contract['adSpaceId'],
        );
        transaction.update(_users.doc(oldResponsibleId), {
          'assignedContractIds': FieldValue.arrayRemove([contractId]),
          if (!hasCustomer)
            'assignedCustomerIds': FieldValue.arrayRemove([
              contract['customerId'],
            ]),
          if (!hasAdSpace)
            'assignedAdSpaceIds': FieldValue.arrayRemove([
              contract['adSpaceId'],
            ]),
          if (previousProfile['role'] == AppRole.supervisor.name)
            'lastAssignedContractId': FieldValue.delete(),
        });
      }
    });
  }

class BusinessDayHelper {
  const BusinessDayHelper._();

  static DateTime nextWorkingDay(DateTime date, List<int> workingDays) {
    if (workingDays.isEmpty ||
        workingDays.any(
          (weekday) => weekday < DateTime.monday || weekday > DateTime.sunday,
        )) {
      throw ArgumentError.value(
        workingDays,
        'workingDays',
        'At least one valid weekday must be selected.',
      );
    }

    final dateOnly = DateTime.utc(date.year, date.month, date.day);
    for (var daysAhead = 0; daysAhead < DateTime.daysPerWeek; daysAhead++) {
      final candidate = dateOnly.add(Duration(days: daysAhead));
      if (workingDays.contains(candidate.weekday)) {
        return DateTime(candidate.year, candidate.month, candidate.day);
      }
    }
    throw StateError('A working day could not be found.');
  }

  static int daysRemaining(DateTime expiryDate, DateTime notificationDate) {
    final expiry = DateTime.utc(
      expiryDate.year,
      expiryDate.month,
      expiryDate.day,
    );
    final notification = DateTime.utc(
      notificationDate.year,
      notificationDate.month,
      notificationDate.day,
    );
    return expiry.difference(notification).inDays;
  }
}

  }) {
    if (count < 1) {

    final targetMonthIndex = startDate.month - 1 + totalMonths;
    final targetYear = startDate.year + targetMonthIndex ~/ 12;
    final targetMonth = targetMonthIndex % 12 + 1;
    final lastDay = DateTime(targetYear, targetMonth + 1, 0).day;

    return DateTime(targetYear, targetMonth, startDate.day.clamp(1, lastDay));
  }

      DateTime(value.year, value.month, value.day);

import 'dart:ffi';
import 'dart:io';
import 'package:ffi/ffi.dart';

typedef _FindWindowWNative =
    IntPtr Function(Pointer<Utf16> lpClassName, Pointer<Utf16> lpWindowName);
typedef _FindWindowWDart =
    int Function(Pointer<Utf16> lpClassName, Pointer<Utf16> lpWindowName);

typedef _IsIconicNative = Int32 Function(IntPtr hWnd);
typedef _IsIconicDart = int Function(int hWnd);

typedef _ShowWindowNative = Int32 Function(IntPtr hWnd, Int32 nCmdShow);
typedef _ShowWindowDart = int Function(int hWnd, int nCmdShow);

typedef _GetForegroundWindowNative = IntPtr Function();
typedef _GetForegroundWindowDart = int Function();

typedef _GetWindowThreadProcessIdNative =
    Uint32 Function(IntPtr hWnd, Pointer<Uint32> lpdwProcessId);
typedef _GetWindowThreadProcessIdDart =
    int Function(int hWnd, Pointer<Uint32> lpdwProcessId);

typedef _GetCurrentThreadIdNative = Uint32 Function();
typedef _GetCurrentThreadIdDart = int Function();

typedef _AttachThreadInputNative =
    Int32 Function(Uint32 idAttach, Uint32 idAttachTo, Int32 fAttach);
typedef _AttachThreadInputDart =
    int Function(int idAttach, int idAttachTo, int fAttach);

typedef _SetForegroundWindowNative = Int32 Function(IntPtr hWnd);
typedef _SetForegroundWindowDart = int Function(int hWnd);

typedef _BringWindowToTopNative = Int32 Function(IntPtr hWnd);
typedef _BringWindowToTopDart = int Function(int hWnd);

class WindowFocusHelper {
  static void bringToFront() {
    if (!Platform.isWindows) return;

    final user32 = DynamicLibrary.open('user32.dll');
    final kernel32 = DynamicLibrary.open('kernel32.dll');

    final findWindow = user32
        .lookupFunction<_FindWindowWNative, _FindWindowWDart>('FindWindowW');
    final isIconic = user32.lookupFunction<_IsIconicNative, _IsIconicDart>(
      'IsIconic',
    );
    final showWindow = user32
        .lookupFunction<_ShowWindowNative, _ShowWindowDart>('ShowWindow');
    final getForegroundWindow = user32
        .lookupFunction<_GetForegroundWindowNative, _GetForegroundWindowDart>(
          'GetForegroundWindow',
        );
    final getWindowThreadProcessId = user32
        .lookupFunction<
          _GetWindowThreadProcessIdNative,
          _GetWindowThreadProcessIdDart
        >('GetWindowThreadProcessId');
    final getCurrentThreadId = kernel32
        .lookupFunction<_GetCurrentThreadIdNative, _GetCurrentThreadIdDart>(
          'GetCurrentThreadId',
        );
    final attachThreadInput = user32
        .lookupFunction<_AttachThreadInputNative, _AttachThreadInputDart>(
          'AttachThreadInput',
        );
    final setForegroundWindow = user32
        .lookupFunction<_SetForegroundWindowNative, _SetForegroundWindowDart>(
          'SetForegroundWindow',
        );
    final bringWindowToTop = user32
        .lookupFunction<_BringWindowToTopNative, _BringWindowToTopDart>(
          'BringWindowToTop',
        );

    // Flutter Windows uygulamasının standart pencere sınıfı adı
    final className = 'FLUTTER_RUNNER_WIN32_WINDOW'.toNativeUtf16();
    final hwnd = findWindow(className, nullptr);
    calloc.free(className);

    if (hwnd == 0) return;

    // 1. Pencere simge durumundaysa (minimized) eski boyutuna döndür (SW_RESTORE = 9)
    if (isIconic(hwnd) != 0) {
      showWindow(hwnd, 9);
    } else {
      // Gizliyse göster (SW_SHOW = 5)
      showWindow(hwnd, 5);
    }

    // 2. Windows'un odak çalma engelini aşarak pencereyi en öne zorla
    final fgHwnd = getForegroundWindow();
    final fgThreadId = fgHwnd != 0
        ? getWindowThreadProcessId(fgHwnd, nullptr)
        : 0;
    final currentThreadId = getCurrentThreadId();

    if (fgThreadId != 0 && fgThreadId != currentThreadId) {
      attachThreadInput(currentThreadId, fgThreadId, 1);
      setForegroundWindow(hwnd);
      bringWindowToTop(hwnd);
      attachThreadInput(currentThreadId, fgThreadId, 0);
    } else {
      setForegroundWindow(hwnd);
      bringWindowToTop(hwnd);
    }
  }
}

  final _mainLayoutKey = GlobalKey<MainLayoutState>();
  late Future<AppUser?> _profile;
  Future<AppStore>? _storeFuture;
  String? _storeUid;
  StreamSubscription<AppUser?>? _profileSubscription;
  String? _invalidUser;

  @override
  void initState() {
    super.initState();
    _profile = _loadProfile(_authService.currentUser);
  }

  Future<AppUser?> _loadProfile(User? user) async {
    if (user == null) return null;
    try {
      final profile = await _authService.loadProfile(user.uid);
      if (profile == null || !profile.isActive) {
        await _authService.signOut();
        if (mounted) setState(() => _invalidUser = 'Hesabınız etkin değil.');
        return null;
      }
      return profile;
    } catch (error) {
      if (mounted) {
        setState(() => _invalidUser = 'Kullanıcı profili alınamadı: $error');
      }
      return null;
    }
  }

  void _onSignedIn(AppUser profile) {
    _profileSubscription?.cancel();
    setState(() {
      _invalidUser = null;
      _profile = Future.value(profile);
      _storeUid = null;
      _storeFuture = null;
    });
  }

  Future<AppStore> _loadStore(AppUser profile) {
    if (_storeFuture == null || _storeUid != profile.uid) {
      _storeUid = profile.uid;
      _storeFuture = _initializeUserStore(profile);
    }
    return _storeFuture!;
  }

  Future<AppStore> _initializeUserStore(AppUser profile) async {
    if (profile.notificationPreferences['push'] != false) {
      try {
        final messaging = FirebaseMessaging.instance;
        await messaging.requestPermission();
        // ignore: deprecated_member_use
        final token = await messaging.getToken(
          vapidKey: kIsWeb && _webVapidKey.isNotEmpty ? _webVapidKey : null,
        );
        if (token != null && token.isNotEmpty) {
          await _authService.registerPushToken(token);
        }
        // ignore: deprecated_member_use
        messaging.onTokenRefresh.listen((token) {
          _authService.registerPushToken(token);
        });
      } catch (_) {
        // Authentication and contract access remain available if push setup is incomplete.
      }
    }
    final store = await AppStore.initialize(
      firestore: widget.firestore,
      user: profile,
    );
    await _profileSubscription?.cancel();
    _profileSubscription = _authService.watchProfile(profile.uid).listen((
      latest,
    ) {
      if (latest == null || !latest.isActive) {
        _authService.signOut();
        if (mounted) {
          setState(() => _invalidUser = 'Hesabınız devre dışı bırakıldı.');
        }
        return;
      }
      final accessChanged =
          latest.role != profile.role ||
          latest.permissions.toString() != profile.permissions.toString() ||
          latest.assignedContractIds.join(',') !=
              profile.assignedContractIds.join(',') ||
          latest.assignedCustomerIds.join(',') !=
              profile.assignedCustomerIds.join(',') ||
          latest.assignedAdSpaceIds.join(',') !=
              profile.assignedAdSpaceIds.join(',');
      if (accessChanged && mounted) {
        setState(() {
          _profile = Future.value(latest);
          _storeUid = null;
          _storeFuture = null;
        });
      }
    }, onError: (_) => _authService.signOut());
    ReminderService.instance.setReviewHandler((contractId) {
      _mainLayoutKey.currentState?.openContract(contractId);
    });
    try {
      await ReminderService.instance.initialize();
      await ReminderService.instance.sync(store);
    } catch (_) {
      // Sign-in remains available if device reminders are not configured.
    }
    store.addListener(() => ReminderService.instance.sync(store));
    return store;
  }

  @override
  void dispose() {
    _profileSubscription?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => StreamBuilder<User?>(
    stream: _authService.authChanges,
    builder: (context, snapshot) {
      if (snapshot.connectionState == ConnectionState.waiting &&
          _authService.currentUser == null) {
        return const Scaffold(body: Center(child: CircularProgressIndicator()));
      }
      final user = snapshot.data ?? _authService.currentUser;
      if (user == null) {
        return LoginScreen(
          authService: _authService,
          errorMessage: _invalidUser,
          onSignedIn: _onSignedIn,
        );
      }
      return FutureBuilder<AppUser?>(
        future: _profile,
        builder: (context, profileSnapshot) {
          if (profileSnapshot.connectionState != ConnectionState.done) {
            return const Scaffold(
              body: Center(child: CircularProgressIndicator()),
            );
          }
          final profile = profileSnapshot.data;
          if (profile == null) {
            return LoginScreen(
              authService: _authService,
              errorMessage: _invalidUser ?? 'Hesap bilgileri bulunamadı.',
              onSignedIn: _onSignedIn,
            );
          }
          return FutureBuilder<AppStore>(
            future: _loadStore(profile),
            builder: (context, storeSnapshot) {
              if (storeSnapshot.connectionState != ConnectionState.done) {
                return const Scaffold(
                  body: Center(child: CircularProgressIndicator()),
                );
              }
              if (storeSnapshot.hasError) {
                return Scaffold(
                  body: Center(
                    child: Text('Veriler yüklenemedi: ${storeSnapshot.error}'),
                  ),
                );
              }
              final store = storeSnapshot.requireData;
              return MainLayout(
                key: _mainLayoutKey,
                store: store,
                user: profile,
                authService: _authService,
              );
            },
          );
        },
      );
    },
  );
}

class ContractPdfService {
  static Future<void> exportContracts(
    AppStore store,
    List<ContractModel> contracts,
  ) async {
    final bytes = await buildContractsPdf(store, contracts);
    final fileName =
        'rst-sozlesmeleri-${DateFormat('yyyyMMdd').format(DateTime.now())}.pdf';
    final file = File(
      '${Directory.systemTemp.path}${Platform.pathSeparator}$fileName',
    );
    await file.writeAsBytes(bytes, flush: true);
    await Process.run('cmd', ['/c', 'start', '', file.path], runInShell: true);
    // await Printing.layoutPdf(name: fileName, onLayout: (_) async => bytes);
  }

  static Future<Uint8List> buildContractsPdf(
    AppStore store,
    List<ContractModel> contracts,
  ) async {
    final fontData = await rootBundle.load(
      'assets/fonts/NotoSans-Variable.ttf',
    );
    final font = pw.Font.ttf(fontData);
    final boldFont = pw.Font.ttf(fontData);
    final document = pw.Document(
      title: 'RST Dolapdere Sözleşme Listesi',
      author: 'RST Dolapdere',
      subject: 'Reklam alanı sözleşmeleri',
    );
    final dateFormat = DateFormat('dd.MM.yyyy');
    final currencyFormat = NumberFormat.currency(
      locale: 'tr_TR',
      symbol: '₺',
      decimalDigits: 2,
    );
    final sorted = [...contracts]
      ..sort((a, b) => a.endDate.compareTo(b.endDate));

    document.addPage(
      pw.MultiPage(
        pageFormat: PdfPageFormat.a4.landscape,
        theme: pw.ThemeData.withFont(base: font, bold: boldFont),
        margin: const pw.EdgeInsets.all(28),
        header: (context) => pw.Padding(
          padding: const pw.EdgeInsets.only(bottom: 14),
          child: pw.Row(
            mainAxisAlignment: pw.MainAxisAlignment.spaceBetween,
            children: [
              pw.Text(
                'RST DOLAPDERE  |  SÖZLEŞMELER',
                style: pw.TextStyle(font: boldFont, fontSize: 16),
              ),
              pw.Text('Dışa aktarma: ${dateFormat.format(DateTime.now())}'),
            ],
          ),
        ),
        footer: (context) => pw.Align(
          alignment: pw.Alignment.centerRight,
          child: pw.Text('${context.pageNumber} / ${context.pagesCount}'),
        ),
        build: (context) => [
          pw.Table(
            border: pw.TableBorder.all(color: PdfColors.grey400, width: 0.5),
            columnWidths: const {
              0: pw.FlexColumnWidth(1.7),
              1: pw.FlexColumnWidth(1.6),
              2: pw.FlexColumnWidth(1.3),
              3: pw.FlexColumnWidth(1.3),
              4: pw.FlexColumnWidth(1.4),
              5: pw.FlexColumnWidth(1.3),
              6: pw.FlexColumnWidth(1.1),
              // 7: pw.FlexColumnWidth(1.2),
            },
            children: [
              _row(
                const [
                  'Reklam alanı',
                  'Kiracı',
                  'Başlangıç',
                  'Bitiş',
                  'Bedel',
                  'Durum',
                  'Süre',
                  'Sorumlu',
                ],
                boldFont,
                header: true,
              ),
              ...sorted.map((contract) {
                final customer = store.customerFor(contract.customerId);
                final space = store.adSpaceFor(contract.adSpaceId);
                final status = contract.isFinalized
                    ? 'Bitti olarak işaretlendi'
                    : contract.endDate.isBefore(DateTime.now())
                    ? 'Süresi doldu'
                    : 'Aktif';
                final durationUnit =
                    contract.durationUnit == ContractDurationUnit.month
                    ? 'ay'
                    : 'yıl';
                return _row([
                  space?.name ?? 'Reklam alanı',
                  customer?.name ?? 'Müşteri',
                  dateFormat.format(contract.startDate),
                  dateFormat.format(contract.endDate),
                  currencyFormat.format(contract.monthlyPrice),
                  status,
                  '${contract.durationCount} $durationUnit',
                  contract.responsibleUsername ?? '-',
                ], font);
              }),
            ],
          ),
        ],
      ),
    );

    return document.save();
  }

  static pw.TableRow _row(
    List<String> values,
    pw.Font font, {
    bool header = false,
  }) => pw.TableRow(
    decoration: header
        ? const pw.BoxDecoration(color: PdfColors.grey200)
        : null,
    children: values
        .map(
          (value) => pw.Padding(
            padding: const pw.EdgeInsets.symmetric(horizontal: 6, vertical: 7),
            child: pw.Text(value, style: pw.TextStyle(font: font, fontSize: 8)),
          ),
        )
        .toList(),
  );

        'sözleşmesi ${DateFormat('dd.MM.yyyy').format(contract.endDate)} tarihinde sona eriyor. '
        '$remainingText';
    final uri = Uri.https('wa.me', '/$phone', {'text': text});
    if (!await launchUrl(uri, mode: LaunchMode.externalApplication)) {

class ReminderService {
  ReminderService._();

  static final ReminderService instance = ReminderService._();
  final FlutterLocalNotificationsPlugin _plugin =
      FlutterLocalNotificationsPlugin();
  bool _ready = false;
  ValueChanged<NotificationResponse>? _onNotificationResponse;

  void setNotificationHandler(ValueChanged<NotificationResponse> handler) {
    _onNotificationResponse = handler;
  }

  void setReviewHandler(ValueChanged<String> handler) {
    WindowFocusHelper.bringToFront(); // <-- Pencereyi öne getirir ve simgeden çıkarır
    _onNotificationResponse = (response) {
      final action = response.actionId ?? '';
      final contractId = action.startsWith('contract:')
          ? action.substring('contract:'.length)
          : response.payload;
      if (contractId != null &&
          contractId.isNotEmpty &&
          !contractId.startsWith('expiring:')) {
        handler(contractId);
      }
    };
  }

  Future<void> initialize() async {
    if (kIsWeb) return;
    final platform = defaultTargetPlatform;
    if (platform != TargetPlatform.android &&
        platform != TargetPlatform.iOS &&
        platform != TargetPlatform.windows) {
      return;
    }
    if (_ready) return;

    timezone_data.initializeTimeZones();
    timezone.setLocalLocation(timezone.getLocation('Europe/Istanbul'));
    await _plugin.initialize(
      settings: _initializationSettings(platform),
      onDidReceiveNotificationResponse: (response) =>
          _onNotificationResponse?.call(response),
    );
    final launchDetails = await _plugin.getNotificationAppLaunchDetails();
    final launchResponse = launchDetails?.notificationResponse;
    if (launchDetails?.didNotificationLaunchApp == true &&
        launchResponse != null) {
      _onNotificationResponse?.call(launchResponse);
    }
    if (platform == TargetPlatform.android) {
      await _plugin
          .resolvePlatformSpecificImplementation<
            AndroidFlutterLocalNotificationsPlugin
          >()
          ?.requestNotificationsPermission();
    }
    if (platform == TargetPlatform.iOS) {
      await _plugin
          .resolvePlatformSpecificImplementation<
            IOSFlutterLocalNotificationsPlugin
          >()
          ?.requestPermissions(alert: true, badge: true, sound: true);
    }
    _ready = true;
  }

  Future<void> snooze(
    String contractId,
    Duration duration,
    List<int> workingDays,
    DateTime endDate,
  ) async {
    if (!_ready) {
      throw StateError('Hatırlatma servisi henüz başlatılmadı.');
    }
    final requestedDate = timezone.TZDateTime.now(timezone.local).add(duration);
    final workingDate = BusinessDayHelper.nextWorkingDay(
      requestedDate,
      workingDays,
    );
    final daysRemaining = BusinessDayHelper.daysRemaining(endDate, workingDate);
    final remainingText = switch (daysRemaining) {
      0 => 'Sözleşme bugün sona eriyor.',
      > 0 => 'Sözleşmenin bitmesine $daysRemaining gün kalıyor.',
      _ => 'Sözleşme ${daysRemaining.abs()} gün önce sona erdi.',
    };
    await _plugin.zonedSchedule(
      id: _snoozedNotificationId(contractId),
      title: 'Ertelenen sözleşme hatırlatması',
      body: remainingText,
      scheduledDate: timezone.TZDateTime(
        timezone.local,
        workingDate.year,
        workingDate.month,
        workingDate.day,
        requestedDate.hour,
        requestedDate.minute,
      ),
      notificationDetails: _notificationDetails(
        defaultTargetPlatform,
        contractId: contractId,
      ),
      androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
      payload: contractId,
    );
  }

  Future<void> sync(AppStore store) async {
    if (!_ready) return;

    final pending = await _plugin.pendingNotificationRequests();
    for (final request in pending) {
      if ((request.id & 0x40000000) == 0) {
        await _plugin.cancel(id: request.id);
      }
    }
    final isWindows = defaultTargetPlatform == TargetPlatform.windows;
    if (isWindows && !store.windowsRemindersEnabled) return;
    if (!isWindows && !store.phoneRemindersEnabled) return;

    final offsets = store.reminderOffsets;
    for (final contract in store.activeContracts) {
      final space = store.adSpaceFor(contract.adSpaceId);
      final customer = store.customerFor(contract.customerId);
      if (space == null) continue;
      final scheduledDates = <String>{};
      for (var index = 0; index < offsets.length; index++) {
        final days = offsets[index];
        final requestedDate = timezone.TZDateTime(
          timezone.local,
          contract.endDate.year,
          contract.endDate.month,
          contract.endDate.day - days,
          store.reminderTimeMinutes ~/ 60,
          store.reminderTimeMinutes % 60,
        );
        final workingDate = BusinessDayHelper.nextWorkingDay(
          requestedDate,
          store.workingDays,
        );
        final reminderDate = timezone.TZDateTime(
          timezone.local,
          workingDate.year,
          workingDate.month,
          workingDate.day,
          requestedDate.hour,
          requestedDate.minute,
        );
        final dateKey =
            '${workingDate.year}-${workingDate.month}-${workingDate.day}';
        if (!scheduledDates.add(dateKey)) continue;
        if (!reminderDate.isAfter(timezone.TZDateTime.now(timezone.local))) {
          continue;
        }
        final daysRemaining = BusinessDayHelper.daysRemaining(
          contract.endDate,
          reminderDate,
        );
        final dayText = switch (daysRemaining) {
          0 => 'bugün sona eriyor',
          > 0 => '$daysRemaining gün sonra sona eriyor',
          _ => '${daysRemaining.abs()} gün önce sona erdi',
        };
        final customerName = customer?.name ?? 'Kiracı';
        await _plugin.zonedSchedule(
          id: _notificationId(contract.id, index),
          title: 'Sözleşme bitişi yaklaşıyor',
          body: '${space.name} alanındaki $customerName sözleşmesi $dayText.',
          scheduledDate: reminderDate,
          notificationDetails: _notificationDetails(
            defaultTargetPlatform,
            contractId: contract.id,
          ),
          androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
          payload: contract.id,
        );
      }
    }
  }

  Future<void> showWindowsTestNotification() async {
    if (!_ready || defaultTargetPlatform != TargetPlatform.windows) return;
    await _plugin.show(
      id: 0x7ffffffe,
      title: 'RST Dolapdere bildirimi',
      body: 'Windows masaüstü bildirimleri etkinleştirildi.',
      notificationDetails: const NotificationDetails(
        windows: WindowsNotificationDetails(
          duration: WindowsNotificationDuration.long,
        ),
      ),
    );
  }

  int _notificationId(String id, int offsetIndex) =>
      (_hashId(id) * 32) + offsetIndex;

  static int _snoozedNotificationId(String id) => 0x40000000 | _hashId(id);

  static int _hashId(String id) {
    var hash = 0;
    for (final unit in id.codeUnits) {
      hash = (hash * 31 + unit) & 0x00ffffff;
    }
    return hash;
  }
}

class _ManagementScreenState extends State<ManagementScreen> {
  final _searchController = TextEditingController();
  String _query = '';
  late ContractListFilter _contractFilter;
  late AdSpaceListFilter _adSpaceFilter;
  bool _nearestFirst = true;
  bool _loadingAvailableContracts = false;
  String? _availableContractsError;
  List<ContractModel> _availableContracts = [];
  final Map<String, String> _availableCustomerNames = {};
  final Map<String, String> _availableAdSpaceNames = {};

  @override
  void initState() {
    super.initState();
    _contractFilter = widget.initialContractFilter;
    _adSpaceFilter = widget.initialAdSpaceFilter;
    if (widget.type == RecordType.contracts &&
        !widget.showEndedContracts &&
        widget.store.currentUser?.role == AppRole.supervisor &&
        widget.authService != null) {
      _loadAvailableContracts();
    }
  }

  String get _title => switch (widget.type) {
    RecordType.adSpaces => 'Reklam Alanları',
    RecordType.customers => 'Müşteriler',
    RecordType.contracts => 'Sözleşmeler',
  };

  List<dynamic> get _records {
    final records = switch (widget.type) {
      RecordType.adSpaces => _adSpaceRecords(),
      RecordType.customers => widget.store.customers,
      RecordType.contracts => _contractRecords(),
    };
    final query = _query.trim().toLowerCase();
    if (query.isEmpty) return records;
    return records
        .where((record) => _searchText(record).contains(query))
        .toList();
  }

  List<AdSpaceModel> _adSpaceRecords() => widget.store.adSpaces.where((space) {
    final occupied = widget.store.activeContractForAdSpace(space.id) != null;
    return switch (_adSpaceFilter) {
      AdSpaceListFilter.all => true,
      AdSpaceListFilter.available => !occupied,
      AdSpaceListFilter.occupied => occupied,
    };
  }).toList();

  List<ContractModel> _contractRecords() {
    final storedContracts = widget.showEndedContracts
        ? widget.store.endedContracts
        : widget.store.contracts
              .where((contract) => !contract.isFinalized)
              .toList();
    final contracts = widget.showEndedContracts
        ? storedContracts
        : [
            ...storedContracts,
            ..._availableContracts.where(
              (available) =>
                  !storedContracts.any((stored) => stored.id == available.id),
            ),
          ];
    if (widget.showEndedContracts) return contracts;
    final filtered = contracts.where((contract) {
      final days = DateHelper.daysUntil(contract.endDate);
      return switch (_contractFilter) {
        ContractListFilter.all => true,
        ContractListFilter.active =>
          widget.store.activeContracts.any(
                (active) => active.id == contract.id,
              ) ||
              (contract.isActive &&
                  !contract.isFinalized &&
                  DateHelper.daysUntil(contract.startDate) <= 0 &&
                  DateHelper.daysUntil(contract.endDate) >= 0),
        ContractListFilter.upcoming => days >= 0 && days <= 10,
        ContractListFilter.expired => days < 0,
      };
    }).toList();
    filtered.sort((a, b) {
      final comparison = a.endDate.compareTo(b.endDate);
      return _nearestFirst ? comparison : -comparison;
    });
    return filtered;
  }

  Future<void> _loadAvailableContracts() async {
    final service = widget.authService;
    if (service == null) return;
    setState(() {
      _loadingAvailableContracts = true;
      _availableContractsError = null;

    try {
      final values = await service.listAvailableContracts();

              ),
            ),
          );
      });

      if (mounted) setState(() => _availableContractsError = error.toString());

      builder: (context) => AlertDialog(
        title: const Text('Sözleşmeyi bitir'),
        content: const Text(
          'Sözleşme kalıcı olarak bitenler listesine alınır. Daha sonra yeniden açılamaz; devam etmek ister misiniz?',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Vazgeç'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Bitir'),
          ),
        ],
      ),
    );
    if (confirmed != true) return;
    await widget.store.markContractEnded(contract.id);
  }

  Future<void> _renewContract(ContractModel contract) =>
      _edit(contract, isRenewal: true);

  Future<void> _showContractDetails(ContractModel contract) => showDialog<void>(
    context: context,
    builder: (context) =>
        _ContractDetailsDialog(store: widget.store, contract: contract),
  );

  Future<void> _showCustomerDetails(CustomerModel customer) => showDialog<void>(
    context: context,
    builder: (context) =>
        _CustomerDetailsDialog(store: widget.store, customer: customer),
  );

  Future<void> _assignResponsible(ContractModel contract) async {
    final service = widget.authService;
    if (service == null) return;
    final user = widget.store.currentUser;
    final userId = user?.role == AppRole.supervisor
        ? user!.uid
        : await showDialog<String>(
            context: context,
            builder: (context) => _AssignResponsibleDialog(
              authService: service,
              currentUserId: user?.uid,
            ),
          );
    if (userId == null) return;
    try {
      await service.assignContract(
        contractId: contract.id,
        responsibleUserId: userId,
      );
      if (user?.role == AppRole.supervisor) {
        setState(() {
          _availableContracts.removeWhere((item) => item.id == contract.id);
        });
        await _loadAvailableContracts();
      }
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(
          context,
        ).showSnackBar(SnackBar(content: Text('Sorumlu atanamadı: $error')));
      }
    }
  }

  Future<void> _toggleContractActive(ContractModel contract) async {
    try {
      await widget.store.setContractActive(contract.id, !contract.isActive);
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Sözleşme durumu değiştirilemedi: $error')),
        );
      }
    }
  }

  Future<void> _exportContracts(List<dynamic> records) async {
    try {
      await ContractPdfService.exportContracts(
        widget.store,
        records.whereType<ContractModel>().toList(),
      );
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(
          context,
        ).showSnackBar(SnackBar(content: Text('PDF oluşturulamadı: $error')));
      }
    }
  }

  Future<void> _exportAdSpaces(List<dynamic> records) async {
    final adSpaces = records.whereType<AdSpaceModel>().toList();

    if (adSpaces.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Dışa aktarılacak reklam alanı bulunamadı.'),
        ),
      );
      return;
    }

    try {
      await AdSpacePdfService.exportAdSpaces(widget.store, adSpaces);
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(
          context,
        ).showSnackBar(SnackBar(content: Text('PDF oluşturulamadı: $error')));
      }
    }
  }

  Future<void> _delete(dynamic record) async {
    final title = _recordTitle(record);
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Kaydı sil'),
        content: Text('$title kaydını silmek istediğinizden emin misiniz?'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Vazgeç'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Sil'),
          ),
        ],
      ),
    );
    if (confirmed != true) return;
    try {
      if (record is AdSpaceModel) {
        await widget.store.deleteAdSpace(record.id);
      } else if (record is CustomerModel) {
        await widget.store.deleteCustomer(record.id);
      } else {
        await widget.store.deleteContract((record as ContractModel).id);
      }
    } on StateError catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(
          context,
        ).showSnackBar(SnackBar(content: Text(error.message)));
      }
    }
  }

  String _recordTitle(dynamic record) {
    if (record is AdSpaceModel) return record.name;
    if (record is CustomerModel) return record.name;
    final contract = record as ContractModel;
    return widget.store.adSpaceFor(contract.adSpaceId)?.name ?? 'Sözleşme';
  }

  Widget _recordTile(dynamic record) {
    String title;
    String subtitle;
    Widget? subtitleWidget;
    IconData icon;
    Color statusColor = Theme.of(context).colorScheme.onSurfaceVariant;
    String? status;

    if (record is AdSpaceModel) {
      final contract = widget.store.activeContractForAdSpace(record.id);
      final customer = contract == null
          ? null
          : widget.store.customerFor(contract.customerId);
      final days = contract == null
          ? null
          : ContractRemainingTime.daysUntil(contract.endDate);
      title = record.name;
      subtitle = contract == null ? 'Kiracı yok' : '';
      if (contract != null) {
        final remainingStyle =
            days != null && ContractRemainingTime.isUrgent(days)
            ? const TextStyle(color: AppColors.danger)
            : null;
        subtitleWidget = Text.rich(
          TextSpan(
            children: [
              TextSpan(text: 'Kiracı: ${customer?.name ?? 'Bilinmiyor'}\n'),
              TextSpan(
                text:
                    'Bitiş: ${DateFormat('dd.MM.yyyy').format(contract.endDate)}\n',
              ),
              const TextSpan(text: 'Kalan süre: '),
              TextSpan(
                text: ContractRemainingTime.format(days!),
                style: remainingStyle,
              ),
            ],
          ),
        );
      }
      icon = Icons.location_city_outlined;
      status = contract == null ? 'Boş' : 'Kirada';
      statusColor = contract == null ? AppColors.success : AppColors.primary;
      if (days != null && ContractRemainingTime.isUrgent(days)) {
        statusColor = AppColors.danger;
      }
    } else if (record is CustomerModel) {
      title = record.name;
      subtitle = [
        record.company,
        record.phone,
        record.email,
      ].where((part) => part.isNotEmpty).join(' • ');
      icon = Icons.person_outline;
      final activeCount = widget.store.activeContracts
          .where((contract) => contract.customerId == record.id)
          .length;
      status = '$activeCount aktif alan';
    } else {
      final contract = record as ContractModel;
      final customer = widget.store.customerFor(contract.customerId);
      final space = widget.store.adSpaceFor(contract.adSpaceId);
      final days = DateHelper.daysUntil(contract.endDate);
      title =
          space?.name ?? _availableAdSpaceNames[contract.id] ?? 'Reklam alanı';
      subtitle =
          '${customer?.name ?? _availableCustomerNames[contract.id] ?? 'Kiracı'} • '
          '${DateFormat('dd.MM.yyyy').format(contract.startDate)} - '
          '${DateFormat('dd.MM.yyyy').format(contract.endDate)} • '
          '${NumberFormat.currency(locale: 'tr_TR', symbol: '₺').format(contract.monthlyPrice)} / ay'
          '${contract.responsibleUsername == null ? '' : ' • Sorumlu: ${contract.responsibleUsername}'}';
      icon = Icons.description_outlined;
      status = contract.isFinalized
          ? 'Bitti olarak işaretlendi'
          : days < 0
          ? 'Süresi doldu'
          : days <= 10
          ? '$days gün kaldı'
          : 'Aktif';
      statusColor = contract.isFinalized || days < 0
          ? Theme.of(context).colorScheme.onSurfaceVariant
          : days <= 10
          ? AppColors.danger
          : AppColors.success;
    }

    return Card(
      child: ListTile(
        contentPadding: const EdgeInsets.symmetric(horizontal: 18, vertical: 8),
        leading: CircleAvatar(
          backgroundColor: Theme.of(
            context,
          ).colorScheme.surfaceContainerHighest,
          child: Icon(icon, color: AppColors.primary),
        ),
        title: Text(title, style: const TextStyle(fontWeight: FontWeight.w600)),
        subtitle: Padding(
          padding: const EdgeInsets.only(top: 5),
          child:
              subtitleWidget ??
              Text(subtitle.isEmpty ? 'Ek bilgi bulunmuyor' : subtitle),
        ),
        trailing: SizedBox(
          width: 130,
          child: Row(
            mainAxisAlignment: MainAxisAlignment.end,
            children: [
              Flexible(
                child: Text(
                  status,
                  textAlign: TextAlign.end,
                  style: TextStyle(
                    color: statusColor,
                    fontSize: 12,
                    fontWeight: FontWeight.w600,
                  ),
                ),
              ),
              PopupMenuButton<String>(
                tooltip: 'Kayıt işlemleri',
                onSelected: (value) {
                  if (record is ContractModel) {
                    switch (value) {
                      case 'details':
                        _showContractDetails(record);
                      case 'toggle':
                        _toggleContractActive(record);
                      case 'assign_self':
                        _assignResponsible(record);
                      case 'edit':
                        _edit(record);
                      case 'end':
                        _markContractEnded(record);
                      case 'renew':
                        _renewContract(record);
                      case 'delete':
                        _delete(record);
                      case 'assign':
                        _assignResponsible(record);
                    }
                  } else if (record is CustomerModel && value == 'details') {
                    _showCustomerDetails(record);
                  } else if (value == 'edit') {
                    _edit(record);
                  } else {
                    _delete(record);
                  }
                },
                itemBuilder: (context) {
                  final supervisor =
                      widget.store.currentUser?.role == AppRole.supervisor;
                  if (record is ContractModel) {
                    final ended =
                        record.isFinalized ||
                        DateHelper.daysUntil(record.endDate) < 0;
                    final supervisor =
                        widget.store.currentUser?.role == AppRole.supervisor;
                    return [
                      const PopupMenuItem(
                        value: 'details',
                        child: Text('Detayları görüntüle'),
                      ),
                      if (supervisor &&
                          record.responsibleUserId ==
                              widget.store.currentUser?.uid)
                        PopupMenuItem(
                          value: 'toggle',
                          child: Text(
                            record.isActive ? 'Pasife al' : 'Etkinleştir',
                          ),
                        ),
                      if (supervisor && record.responsibleUserId == null)
                        const PopupMenuItem(
                          value: 'assign_self',
                          child: Text('Kendime ata'),
                        ),
                      if (!supervisor && !ended)
                        const PopupMenuItem(
                          value: 'edit',
                          child: Text('Düzenle'),
                        ),
                      if (!supervisor && !record.isFinalized)
                        const PopupMenuItem(
                          value: 'end',
                          child: Text('Bitti olarak işaretle'),
                        ),
                      const PopupMenuItem(
                        value: 'renew',
                        child: Text('Sözleşmeyi uzat'),
                      ),
                      if (!supervisor &&
                          (widget.store.currentUser?.role == AppRole.admin ||
                              widget.store.currentUser?.role ==
                                  AppRole.manager))
                        const PopupMenuItem(
                          value: 'assign',
                          child: Text('Sorumlu ata'),
                        ),
                      if (!supervisor)
                        const PopupMenuItem(
                          value: 'delete',
                          child: Text('Sil'),
                        ),
                    ];
                  }
                  if (supervisor && record is AdSpaceModel) {
                    return const [];
                  }
                  if (supervisor && record is CustomerModel) {
                    return const [
                      PopupMenuItem(
                        value: 'details',
                        child: Text('Detayları görüntüle'),
                      ),
                    ];
                  }
                  return [
                    if (record is CustomerModel)
                      const PopupMenuItem(
                        value: 'details',
                        child: Text('Detayları görüntüle'),
                      ),
                    const PopupMenuItem(value: 'edit', child: Text('Düzenle')),
                    const PopupMenuItem(value: 'delete', child: Text('Sil')),
                  ];
                },
              ),
            ],
          ),
        ),
      ),
    );
  }

class _AssignResponsibleDialog extends StatefulWidget {
  const _AssignResponsibleDialog({
    required this.authService,
    this.currentUserId,
  });

  final AuthService authService;
  final String? currentUserId;

  @override
  State<_AssignResponsibleDialog> createState() =>
      _AssignResponsibleDialogState();
}

class _AssignResponsibleDialogState extends State<_AssignResponsibleDialog> {
  String? _selectedUserId;

  @override
  Widget build(BuildContext context) => AlertDialog(
    title: const Text('Sözleşme sorumlusu'),
    content: SizedBox(
      width: 380,
      child: StreamBuilder<List<AppUser>>(
        stream: widget.authService.watchUsers(),
        builder: (context, snapshot) {
          final users = (snapshot.data ?? const <AppUser>[])
              .where((user) => user.isActive)
              .toList();
          if (snapshot.hasError) {
            return Text('Kullanıcılar alınamadı: ${snapshot.error}');
          }
          if (users.isEmpty) return const Text('Etkin kullanıcı bulunamadı.');
          _selectedUserId ??=
              users
                  .where((user) => user.uid == widget.currentUserId)
                  .firstOrNull
                  ?.uid ??
              users.first.uid;
          return DropdownButtonFormField<String>(
            initialValue: _selectedUserId,
            decoration: const InputDecoration(labelText: 'Sorumlu kullanıcı'),
            items: users
                .map(
                  (user) => DropdownMenuItem(
                    value: user.uid,
                    child: Text('${user.username} (${user.role.name})'),
                  ),
                )
                .toList(),
            onChanged: (value) => setState(() => _selectedUserId = value),
          );
        },
      ),
    ),
    actions: [
      TextButton(
        onPressed: () => Navigator.pop(context),
        child: const Text('Vazgeç'),
      ),
      FilledButton(
        onPressed: _selectedUserId == null
            ? null
            : () => Navigator.pop(context, _selectedUserId),
        child: const Text('Ata'),
      ),
    ],
  );
}

class _ContractDetailsDialog extends StatelessWidget {
  const _ContractDetailsDialog({required this.store, required this.contract});

  final AppStore store;
  final ContractModel contract;

  @override
  Widget build(BuildContext context) {
    final dateFormat = DateFormat('dd.MM.yyyy');
    final customer = store.customerFor(contract.customerId);
    final space = store.adSpaceFor(contract.adSpaceId);
    final history = [...contract.history]
      ..sort((a, b) => a.occurredAt.compareTo(b.occurredAt));
    final currencyFormat = NumberFormat.currency(locale: 'tr_TR', symbol: '₺');

    String actionLabel(ContractHistoryAction action) => switch (action) {
      ContractHistoryAction.created => 'Sözleşme eklendi',
      ContractHistoryAction.renewed => 'Sözleşme uzatıldı',
      ContractHistoryAction.ended => 'Sözleşme bitirildi',
      ContractHistoryAction.activated => 'Sözleşme etkinleştirildi',
      ContractHistoryAction.assigned => 'Sözleşme sorumlusu atandı',
    };

    return AlertDialog(
      title: Text(space?.name ?? 'Sözleşme detayı'),
      content: SizedBox(
        width: 520,
        child: SingleChildScrollView(
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            mainAxisSize: MainAxisSize.min,
            children: [
              Text('Kiracı: ${customer?.name ?? 'Bilinmiyor'}'),
              Text('Alan: ${space?.name ?? 'Silinmiş reklam alanı'}'),
              Text('Başlangıç: ${dateFormat.format(contract.startDate)}'),
              Text('Bitiş: ${dateFormat.format(contract.endDate)}'),
              Text(
                'Aylık bedel: ${NumberFormat.currency(locale: 'tr_TR', symbol: '₺').format(contract.monthlyPrice)}',
              ),
              if (contract.notes.isNotEmpty) Text('Notlar: ${contract.notes}'),
              const SizedBox(height: 20),
              Text(
                'Hareketler',
                style: Theme.of(context).textTheme.titleMedium,
              ),
              const SizedBox(height: 8),
              if (history.isEmpty)
                const Text('Bu sözleşme için kayıtlı hareket bulunmuyor.'),
              ...history.map((entry) {
                final detail = switch (entry.action) {
                  ContractHistoryAction.created =>
                    entry.newEndDate == null
                        ? ''
                        : 'Bitiş: ${dateFormat.format(entry.newEndDate!)}'
                              '${entry.newMonthlyPrice == null ? '' : ' • Bedel: ${currencyFormat.format(entry.newMonthlyPrice)}'}',
                  ContractHistoryAction.renewed =>
                    '${entry.previousEndDate == null ? '-' : dateFormat.format(entry.previousEndDate!)} → '
                        '${entry.newEndDate == null ? '-' : dateFormat.format(entry.newEndDate!)}'
                        '${entry.previousMonthlyPrice == null || entry.newMonthlyPrice == null ? '' : ' • Bedel: ${currencyFormat.format(entry.previousMonthlyPrice)} → ${currencyFormat.format(entry.newMonthlyPrice)}'}',
                  ContractHistoryAction.ended =>
                    'Bitiş: ${dateFormat.format(entry.newEndDate ?? contract.endDate)}',
                  ContractHistoryAction.activated =>
                    'Bitiş: ${dateFormat.format(entry.newEndDate ?? contract.endDate)}',
                  ContractHistoryAction.assigned =>
                    'Sorumlu: ${entry.targetUsername ?? contract.responsibleUsername ?? '-'}',
                };
                return ListTile(
                  contentPadding: EdgeInsets.zero,
                  dense: true,
                  leading: const Icon(Icons.history),
                  title: Text(actionLabel(entry.action)),
                  subtitle: Text(
                    '${dateFormat.format(entry.occurredAt)}'
                    '${entry.actorUsername == null ? '' : ' • İşlemi yapan: ${entry.actorUsername}'}'
                    '${detail.isEmpty ? '' : ' • $detail'}',
                  ),
                );
              }),
            ],
          ),
        ),
      ),
      actions: [

class _RecordFormDialogState extends State<_RecordFormDialog> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _name;
  late final TextEditingController _company;
  late final TextEditingController _phone;
  late final TextEditingController _email;
  late final TextEditingController _location;
  late final TextEditingController _description;
  late final TextEditingController _price;
  late final TextEditingController _notes;
  late final TextEditingController _durationCount;
  late String? _customerId;
  late String? _adSpaceId;

      text: initial is AdSpaceModel ? initial.description : '',

          : initial is ContractModel
          ? initial.monthlyPrice.toString()
          : '',
    );
    _notes = TextEditingController(
      text: initial is CustomerModel
          ? initial.notes
          : initial is ContractModel
          ? initial.notes
          : '',
    );
    _durationUnit = initial is ContractModel
        ? initial.durationUnit
        : ContractDurationUnit.year;
    _durationCount = TextEditingController(
      text: initial is ContractModel ? '${initial.durationCount}' : '1',
    );
    _customerId = initial is ContractModel

        : widget.store.customers.firstOrNull?.id;
    _adSpaceId = initial is ContractModel
        ? initial.adSpaceId
        : widget.store.adSpaces.firstOrNull?.id;

      initial is ContractModel ? initial.startDate : DateTime.now(),
    );

      initial is ContractModel
          ? widget.isRenewal
                ? ContractDateHelper.addDuration(
                    initial.endDate,
                    unit: _durationUnit,
                    count: int.tryParse(_durationCount.text) ?? 1,
                  )
                : initial.endDate
          : ContractDateHelper.addDuration(
              _startDate,
              unit: _durationUnit,
              count: int.tryParse(_durationCount.text) ?? 1,
            ),
    );
    _active = widget.isRenewal
        ? true
        : initial is ContractModel
        ? initial.isActive
        : true;
  }

  @override
  void dispose() {
    for (final controller in [
      _name,
      _company,
      _phone,
      _email,
      _location,
      _description,
      _price,
      _notes,
      _durationCount,
    ]) {
      controller.dispose();
    }
    super.dispose();
  }

  String get _title => widget.isRenewal
      ? 'Sözleşmeyi uzat'
      : widget.initial == null
      ? 'Yeni kayıt'
      : 'Kaydı düzenle';

  void _updateStartDate(DateTime value) {
    setState(() {
      _startDate = value;
      _recalculateEndDate();
    });
  }

  void _updateEndDate(DateTime value) {
    setState(() => _endDate = value);
  }

  void _recalculateEndDate() {
    final count = int.tryParse(_durationCount.text.trim());
    if (count == null || count < 1) return;
    final extensionBase = widget.isRenewal && widget.initial is ContractModel
        ? (widget.initial as ContractModel).endDate
        : _startDate;
    _endDate = ContractDateHelper.addDuration(
      extensionBase,
      unit: _durationUnit,
      count: count,
    );
  }

  void _setDurationUnit(ContractDurationUnit unit) {
    setState(() {
      _durationUnit = unit;
      _recalculateEndDate();
    });
  }

  void _setDurationCount(String value) {
    final count = int.tryParse(value);
    if (count == null || count < 1 || count > 99) return;
    setState(_recalculateEndDate);
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    if (widget.store.currentUser?.role == AppRole.supervisor &&
        widget.type == RecordType.contracts &&
        _parsePrice() <= 0) {
      setState(() => _error = 'Birim fiyat sıfırdan büyük olmalıdır.');
      return;
    }
    final id = widget.isRenewal
        ? DateTime.now().microsecondsSinceEpoch.toString()
        : switch (widget.initial) {
            AdSpaceModel value => value.id,
            CustomerModel value => value.id,
            ContractModel value => value.id,
            _ => DateTime.now().microsecondsSinceEpoch.toString(),
          };
    try {
      if (widget.type == RecordType.adSpaces) {
        await widget.store.saveAdSpace(
          AdSpaceModel(
            id: id,
            name: _name.text.trim(),
            location: _location.text.trim(),
            description: _description.text.trim(),
            monthlyPrice: _parsePrice(),
            isAvailable: widget.initial is AdSpaceModel
                ? widget.initial.isAvailable
                : true,
          ),
        );
      } else if (widget.type == RecordType.customers) {
        await widget.store.saveCustomer(
          CustomerModel(
            id: id,
            name: _name.text.trim(),
            company: _company.text.trim(),
            phone: _phone.text.trim(),
            email: _email.text.trim(),
            notes: _notes.text.trim(),
          ),
        );
      } else {
        final contract = ContractModel(
          id: id,
          customerId: _customerId!,
          adSpaceId: _adSpaceId!,
          startDate: _startDate,
          endDate: _endDate,
          monthlyPrice: _parsePrice(),
          durationUnit: _durationUnit,
          durationCount: int.parse(_durationCount.text),
          notes: _notes.text.trim(),
          isActive: _active,
        );
        if (widget.isRenewal && widget.initial is ContractModel) {
          await widget.store.renewContract(
            (widget.initial as ContractModel).id,
            contract,
          );
        } else {
          await widget.store.saveContract(contract);
        }
      }
      if (mounted) Navigator.pop(context, true);
    } on StateError catch (error) {
      setState(() => _error = error.message);
    } catch (error) {
      if (mounted) setState(() => _error = 'İşlem tamamlanamadı: $error');
    }
  }

  double _parsePrice() =>
      double.tryParse(_price.text.trim().replaceAll(',', '.')) ?? 0;

class _DatePartsField extends StatefulWidget {
  const _DatePartsField({
    required this.label,
    required this.initialDate,
    required this.onChanged,
    this.readOnly = false,
    this.validator,
  });

  final String label;
  final DateTime initialDate;
  final ValueChanged<DateTime> onChanged;
  final bool readOnly;
  final String? Function(DateTime)? validator;

  @override
  State<_DatePartsField> createState() => _DatePartsFieldState();
}

class _DatePartsFieldState extends State<_DatePartsField> {
  late final TextEditingController _day;
  late final TextEditingController _month;
  late final TextEditingController _year;

  @override
  void initState() {
    super.initState();
    _day = TextEditingController();
    _month = TextEditingController();
    _year = TextEditingController();
    _setText(widget.initialDate);
  }

  @override
  void didUpdateWidget(covariant _DatePartsField oldWidget) {
    super.didUpdateWidget(oldWidget);
    final newDate = ContractDateHelper.dateOnly(widget.initialDate);
    if (ContractDateHelper.dateOnly(oldWidget.initialDate) != newDate) {
      if (_readDate() == newDate) return;
      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (!mounted ||
            ContractDateHelper.dateOnly(widget.initialDate) != newDate) {
          return;
        }
        _setText(newDate);
      });
    }
  }

  @override
  void dispose() {
    _day.dispose();
    _month.dispose();
    _year.dispose();
    super.dispose();
  }

  void _setText(DateTime date) {
    _day.text = date.day.toString().padLeft(2, '0');
    _month.text = date.month.toString().padLeft(2, '0');
    _year.text = date.year.toString().padLeft(4, '0');
  }

  DateTime? _readDate() {
    final day = int.tryParse(_day.text);
    final month = int.tryParse(_month.text);
    final year = int.tryParse(_year.text);
    if (day == null ||
        month == null ||
        year == null ||
        year < 2000 ||
        year > 2100) {
      return null;
    }
    if (month < 1 || month > 12) return null;
    final lastDay = DateTime(year, month + 1, 0).day;
    if (day < 1 || day > lastDay) return null;
    return DateTime(year, month, day);
  }

  void _handleInput(FormFieldState<DateTime> fieldState) {
    final value = _readDate();
    fieldState.didChange(value);
    if (value != null) widget.onChanged(value);
  }

  Future<void> _pickDate(FormFieldState<DateTime> fieldState) async {
    final current = _readDate() ?? widget.initialDate;
    final picked = await showDatePicker(
      context: context,
      locale: const Locale('tr', 'TR'),
      initialDate: current,
      firstDate: DateTime(2000),
      lastDate: DateTime(2100, 12, 31),
    );
    if (picked == null || !mounted) return;
    _setText(picked);
    fieldState.didChange(picked);
    widget.onChanged(ContractDateHelper.dateOnly(picked));
  }

  Widget _partInput({
    required String hint,
    required TextEditingController controller,
    required int maxLength,
    required FormFieldState<DateTime> fieldState,
  }) => Expanded(
    flex: hint == 'YYYY' ? 3 : 2,
    child: TextFormField(
      controller: controller,
      textAlign: TextAlign.center,
      keyboardType: TextInputType.number,
      enabled: !widget.readOnly,
      inputFormatters: [
        FilteringTextInputFormatter.digitsOnly,
        LengthLimitingTextInputFormatter(maxLength),
      ],
      decoration: InputDecoration(
        hintText: hint,
        counterText: '',
        isDense: true,
        filled: true,
        contentPadding: const EdgeInsets.symmetric(horizontal: 6, vertical: 14),
        border: OutlineInputBorder(borderRadius: BorderRadius.circular(8)),
      ),
      onChanged: (_) => _handleInput(fieldState),
    ),
  );

  @override
  Widget build(BuildContext context) => FormField<DateTime>(
    initialValue: widget.initialDate,
    autovalidateMode: AutovalidateMode.onUserInteraction,
    validator: (_) {
      final date = _readDate();
      if (date == null) return 'GG / AA / YYYY biçiminde geçerli tarih girin.';
      return widget.validator?.call(date);
    },
    builder: (fieldState) => Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(widget.label, style: Theme.of(context).textTheme.labelLarge),
        const SizedBox(height: 8),
        Row(
          children: [
            _partInput(
              hint: 'GG',
              controller: _day,
              maxLength: 2,
              fieldState: fieldState,
            ),
            const SizedBox(width: 8),
            _partInput(
              hint: 'AA',
              controller: _month,
              maxLength: 2,
              fieldState: fieldState,
            ),
            const SizedBox(width: 8),
            _partInput(
              hint: 'YYYY',
              controller: _year,
              maxLength: 4,
              fieldState: fieldState,
            ),
            const SizedBox(width: 4),
            IconButton(
              tooltip: 'Takvimden seç',
              onPressed: widget.readOnly ? null : () => _pickDate(fieldState),
              icon: const Icon(Icons.calendar_month_outlined),
            ),
          ],
        ),
        if (fieldState.errorText != null)
          Padding(
            padding: const EdgeInsets.only(top: 6),
            child: Text(
              fieldState.errorText!,
              style: TextStyle(
                color: Theme.of(context).colorScheme.error,
                fontSize: 12,
              ),
            ),
          ),
        const SizedBox(height: 12),
      ],
    ),
  );
}

  static Future<void> exportAdSpaces(
    AppStore store,
    List<AdSpaceModel> adSpaces,
  ) async {
    final bytes = await buildAdSpacesPdf(store, adSpaces);

    final fileName =
        'rst-reklam-alanlari-${DateTime.now().millisecondsSinceEpoch}.pdf';

    final file = File(
      '${Directory.systemTemp.path}${Platform.pathSeparator}$fileName',
    );

    await file.writeAsBytes(bytes, flush: true);

    await Process.run('cmd', ['/c', 'start', '', file.path], runInShell: true);
  }

  static Future<Uint8List> buildAdSpacesPdf(
    AppStore store,
    List<AdSpaceModel> adSpaces,
  ) async {
    final fontData = await rootBundle.load(
      'assets/fonts/NotoSans-Variable.ttf',
    );
    final font = pw.Font.ttf(fontData);
    final boldFont = pw.Font.ttf(fontData);
    final dateFormat = DateFormat('dd.MM.yyyy');
    final document = pw.Document(
      title: 'RST Dolapdere Reklam Alanları',
      author: 'RST Dolapdere',
      subject: 'Reklam alanı ve kiracı durumu',
    );
    final sorted = [...adSpaces]..sort((a, b) => a.name.compareTo(b.name));

    document.addPage(
      pw.MultiPage(
        pageFormat: PdfPageFormat.a4.landscape,
        theme: pw.ThemeData.withFont(base: font, bold: boldFont),
        margin: const pw.EdgeInsets.all(28),

              ),
              pw.Text('Dışa aktarma: ${dateFormat.format(DateTime.now())}'),
            ],
          ),
        ),
        footer: (context) => pw.Align(
          alignment: pw.Alignment.centerRight,
          child: pw.Text('${context.pageNumber} / ${context.pagesCount}'),
        ),
        build: (context) => [
          pw.Table(
            border: pw.TableBorder.all(color: PdfColors.grey400, width: 0.5),
            columnWidths: const {
              0: pw.FlexColumnWidth(2),
              1: pw.FlexColumnWidth(2),
              2: pw.FlexColumnWidth(1.3),
              3: pw.FlexColumnWidth(1.2),
              4: pw.FlexColumnWidth(1.2),

                  'Durum',
                  'Sorumlu',

                boldFont,
                header: true,
              ),
              ...sorted.map((space) {

                final customer = contract == null
                    ? null
                    : store.customerFor(contract.customerId);
                final days = contract == null
                    ? null
                    : ContractRemainingTime.daysUntil(contract.endDate);

                        : dateFormat.format(contract.endDate),
                    days == null ? '-' : ContractRemainingTime.format(days),
                    contract == null ? 'Boş' : 'Kirada',
                    contract?.responsibleUsername ?? '-',
                  ],
                  font,
                  remainingColor:
                      days != null && ContractRemainingTime.isUrgent(days)
                      ? PdfColors.red
                      : null,
                );
              }),
            ],
          ),
        ],
      ),
    );

    return document.save();
  }

  static pw.TableRow _row(
    List<String> values,
    pw.Font font, {
    bool header = false,
    PdfColor? remainingColor,
  }) => pw.TableRow(
    decoration: header
        ? const pw.BoxDecoration(color: PdfColors.grey200)
        : null,
    children: values.indexed
        .map(
          (entry) => pw.Padding(
            padding: const pw.EdgeInsets.symmetric(horizontal: 6, vertical: 7),
            child: pw.Text(
              entry.$2,
              style: pw.TextStyle(
                font: font,
                fontSize: 8,
                color: entry.$1 == 3 ? remainingColor : null,
              ),
            ),
          ),
        )
        .toList(),
  );
}

    bool? phone,

    int? reminderMinutes,
    List<int>? reminderOffsets,
    List<int>? workingDays,
  }) => store.saveReminderSettings(
    phone: phone ?? store.phoneRemindersEnabled,
    windows: windows,
    reminderMinutes: reminderMinutes,
    reminderOffsets: reminderOffsets,
    workingDays: workingDays,
  );

  Future<void> _setReminderOffset(int offset, bool selected) async {
    final next = [...store.reminderOffsets];
    if (selected) {
      next.add(offset);
    } else {
      next.remove(offset);
    }
    if (next.isEmpty) {
      _showMessage('En az bir bildirim günü seçmelisiniz.');
      return;
    }
    await _save(reminderOffsets: next);
  }

  Future<void> _setWorkingDay(int weekday, bool selected) async {
    final next = [...store.workingDays];
    if (selected) {
      next.add(weekday);
    } else {
      next.remove(weekday);
    }
    if (next.isEmpty) {
      _showMessage('En az bir çalışma günü seçmelisiniz.');
      return;
    }
    await _save(workingDays: next);
  }

  Future<void> _setWindowsNotifications(bool enabled) async {
    if (!enabled) {
      await _save(windows: false);
      return;
    }
    try {
      final opened = await launchUrl(
        Uri.parse('ms-settings:notifications'),
        mode: LaunchMode.externalApplication,
      );
      if (!opened) {
        throw StateError('Windows bildirim ayarları açılamadı.');
      }
      await ReminderService.instance.initialize();
      await ReminderService.instance.showWindowsTestNotification();
      await _save(windows: true);
      if (mounted) {
        _showMessage(
          'Windows bildirim ayarlarında RST Dolapdere bildirimlerini ve başlıkları etkinleştirin.',
        );
      }
    } catch (error) {
      _showMessage('Windows bildirimleri etkinleştirilemedi: $error');
    }
  }

    final initialTime = TimeOfDay(

      context: context,
      initialTime: initialTime,
      helpText: 'Bildirim saatini seçin',
    );
    if (selected == null) return;
    await _save(reminderMinutes: selected.hour * 60 + selected.minute);

        if (permission.authorizationStatus == AuthorizationStatus.denied) {

          await authService.registerPushToken(token);
        } else {
          throw StateError('Push tokenı alınamadı.');
        }
      }
      await authService.updateOwnNotificationPreferences(next);

        _showMessage('Bildirim tercihi kaydedilemedi: $error');

class _ChangePasswordDialogState extends State<_ChangePasswordDialog> {
  final _formKey = GlobalKey<FormState>();
  final _current = TextEditingController();
  final _next = TextEditingController();

  @override
  void dispose() {
    _current.dispose();
    _next.dispose();
    super.dispose();
  }

  void _submit() {
    if (!_formKey.currentState!.validate()) return;
    Navigator.of(context).pop((current: _current.text, next: _next.text));
  }

  @override
  Widget build(BuildContext context) => AlertDialog(
    title: const Text('Şifre değiştir'),
    content: SizedBox(
      width: 400,
      child: Form(
        key: _formKey,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            TextFormField(
              controller: _current,
              obscureText: true,
              decoration: const InputDecoration(labelText: 'Mevcut şifre'),
              validator: (value) => value == null || value.isEmpty
                  ? 'Mevcut şifrenizi girin.'
                  : null,
            ),
            TextFormField(
              controller: _next,
              obscureText: true,
              onFieldSubmitted: (_) => _submit(),
              decoration: const InputDecoration(
                labelText: 'Yeni şifre',
                helperText: 'En az 10 karakter',
              ),
              validator: (value) => value == null || value.length < 10
                  ? 'En az 10 karakter girin.'
                  : null,
            ),
          ],
        ),
      ),
    ),
    actions: [
      TextButton(
        onPressed: () => Navigator.of(context).pop(),
        child: const Text('Vazgeç'),
      ),
      FilledButton(onPressed: _submit, child: const Text('Değiştir')),

  );

class MainLayoutState extends State<MainLayout> {
  int selectedIndex = 0;
  ContractListFilter _contractFilter = ContractListFilter.all;
  AdSpaceListFilter _adSpaceFilter = AdSpaceListFilter.all;
  String? _pendingContractId;

  final List<String> pageTitles = [
    'Ana Panel',
    'Reklam Alanları',
    'Müşteriler',
    'Sözleşmeler',
    'Biten Sözleşmeler',
    'Bildirimler',
    'Ayarlar',
  ];

  bool _canAccess(String module) => widget.user?.canAccess(module) ?? true;

  List<int> get _visibleIndices => [
    if (_canAccess('dashboard')) 0,
    if (_canAccess('adSpaces')) 1,
    if (_canAccess('customers')) 2,
    if (_canAccess('contracts')) ...[3, 4],
    if (_canAccess('notifications')) 5,
    if (_canAccess('settings')) 6,
  ];

  @override
  void initState() {
    super.initState();
    if (!_canAccess('dashboard') && _visibleIndices.isNotEmpty) {
      selectedIndex = _visibleIndices.first;
    }
    ReminderService.instance.setNotificationHandler(
      _handleNotificationResponse,
    );
    widget.store.addListener(_onStoreChanged);
  }

  void initializeReminders() {
    unawaited(_initializeReminders());
  }

  Future<void> _initializeReminders() async {
    try {
      await ReminderService.instance.initialize();
      await ReminderService.instance.sync(widget.store);
    } catch (error) {
      debugPrint('Masaüstü bildirimleri başlatılamadı: $error');
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('Masaüstü bildirimleri başlatılamadı: $error'),
          ),
        );
      }
    }
  }

  void _onStoreChanged() {
    final pendingContractId = _pendingContractId;
    if (pendingContractId != null &&
        widget.store.contracts.any(
          (contract) => contract.id == pendingContractId,
        )) {
      _pendingContractId = null;
      openContract(pendingContractId);
    }
    unawaited(_syncReminders());
  }

  Future<void> _syncReminders() async {
    try {
      await ReminderService.instance.sync(widget.store);
    } catch (error) {
      debugPrint('Masaüstü bildirimleri eşitlenemedi: $error');
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Masaüstü bildirimleri eşitlenemedi: $error')),
        );
      }
    }
  }

  void _handleNotificationResponse(NotificationResponse response) {
    WindowFocusHelper.bringToFront(); // <-- Pencereyi öne getirir ve simgeden çıkarır
    final action = response.actionId ?? '';
    if (action == 'snooze' || action.startsWith('snooze:')) {
      final contractId = action.startsWith('snooze:')
          ? action.substring('snooze:'.length)
          : response.payload;
      if (contractId != null && contractId.isNotEmpty) {
        unawaited(_chooseSnoozeDuration(contractId));
      }
      return;
    }
    if (action == 'expiring' || action.startsWith('expiring:')) {
      openExpiringContracts();
      return;
    }
    if (action.startsWith('contract:')) {
      _openNotifiedContract(action.substring('contract:'.length));
      return;
    }
    final contractId = response.payload;
    if (contractId != null && contractId.isNotEmpty) {
      _openNotifiedContract(contractId);
    }
  }

  Future<void> _chooseSnoozeDuration(String contractId) async {
    final duration = await showDialog<Duration>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Hatırlatmayı ertele'),
        content: SingleChildScrollView(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              for (final (label, duration) in const [
                ('5 dakika', Duration(minutes: 5)),
                ('10 dakika', Duration(minutes: 10)),
                ('30 dakika', Duration(minutes: 30)),
                ('1 saat', Duration(hours: 1)),
                ('2 saat', Duration(hours: 2)),
                ('1 gün', Duration(days: 1)),
                ('2 gün', Duration(days: 2)),
              ])
                ListTile(
                  title: Text(label),
                  leading: const Icon(Icons.snooze_outlined),
                  onTap: () => Navigator.of(context).pop(duration),
                ),
            ],
          ),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(context).pop(),
            child: const Text('Vazgeç'),
          ),
        ],
      ),
    );
    if (duration == null || !mounted) return;
    try {
      final contracts = widget.store.contracts.where(
        (item) => item.id == contractId,
      );
      if (contracts.isEmpty) {
        throw StateError('Sözleşme bulunamadı.');
      }
      final contract = contracts.first;
      await ReminderService.instance.snooze(
        contractId,
        duration,
        widget.store.workingDays,
        contract.endDate,
      );
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              'Hatırlatma ${_snoozeDurationLabel(duration)} ertelendi.',
            ),
          ),
        );
      }
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Hatırlatma ertelenemedi: $error')),
        );
      }
    }
  }

  String _snoozeDurationLabel(Duration duration) {
    if (duration.inDays > 0) return '${duration.inDays} gün';
    if (duration.inHours > 0) return '${duration.inHours} saat';
    return '${duration.inMinutes} dakika';
  }

  void _openNotifiedContract(String contractId) {
    if (!_canAccess('contracts')) return;
    if (!widget.store.contracts.any((contract) => contract.id == contractId)) {
      _pendingContractId = contractId;
      return;
    }
    openContract(contractId);
  }

  Widget getSelectedPage() {
    final module = switch (selectedIndex) {
      0 => 'dashboard',
      1 => 'adSpaces',
      2 => 'customers',
      3 || 4 => 'contracts',
      5 => 'notifications',
      6 => 'settings',
      _ => null,
    };
    if (module != null && !_canAccess(module)) {
      return const Center(child: Text('Bu bölümü görüntüleme yetkiniz yok.'));
    }
    switch (selectedIndex) {
      case 0:
        return DashboardScreen(
          store: widget.store,
          onNavigate: _canAccess('dashboard') ? _navigateFromDashboard : null,
        );
      case 1:
        return ManagementScreen(
          store: widget.store,
          type: RecordType.adSpaces,
          initialAdSpaceFilter: _adSpaceFilter,
          authService: widget.authService,
        );
      case 2:
        return ManagementScreen(
          store: widget.store,
          type: RecordType.customers,
          authService: widget.authService,
        );
      case 3:
        return ManagementScreen(
          key: ValueKey('contracts-${_contractFilter.name}'),
          store: widget.store,
          type: RecordType.contracts,
          initialContractFilter: _contractFilter,
          authService: widget.authService,
        );
      case 4:
        return ManagementScreen(
          key: ValueKey('ended-contracts-${_contractFilter.name}'),
          store: widget.store,
          type: RecordType.contracts,
          showEndedContracts: true,
          authService: widget.authService,
        );
      case 5:
        return NotificationScreen(
          store: widget.store,
          user: widget.user,
          onReviewContract: openContract,
        );
      case 6:
        return SettingsScreen(
          store: widget.store,
          user: widget.user,
          authService: widget.authService,
        );

      default:
        return Center(
          child: Text(
            '${pageTitles[selectedIndex]} ekranını sonraki aşamada hazırlayacağız.',
            textAlign: TextAlign.center,
            style: TextStyle(
              fontSize: 16,
              color: Theme.of(context).colorScheme.onSurfaceVariant,
            ),
          ),
        );
    }
  }

  void _selectPage(int index) {
    setState(() {
      selectedIndex = index;
      _contractFilter = ContractListFilter.all;
      _adSpaceFilter = AdSpaceListFilter.all;
    });
  }

  void _navigateFromDashboard(DashboardDestination destination) {
    final requiredModule = switch (destination) {
      DashboardDestination.adSpaces ||
      DashboardDestination.availableAdSpaces => 'adSpaces',
      DashboardDestination.activeContracts ||
      DashboardDestination.expiringContracts => 'contracts',
    };
    if (!_canAccess(requiredModule)) return;
    setState(() {
      switch (destination) {
        case DashboardDestination.adSpaces:
          selectedIndex = 1;
          _adSpaceFilter = AdSpaceListFilter.all;
        case DashboardDestination.activeContracts:
          selectedIndex = 3;
          _contractFilter = ContractListFilter.active;
        case DashboardDestination.availableAdSpaces:
          selectedIndex = 1;
          _adSpaceFilter = AdSpaceListFilter.available;
        case DashboardDestination.expiringContracts:
          selectedIndex = 3;
          _contractFilter = ContractListFilter.upcoming;
      }
    });
  }

  void openContract(String contractId) {
    if (!_canAccess('contracts')) return;
    final contract = widget.store.contracts.where(
      (item) => item.id == contractId,
    );
    if (contract.isEmpty) return;
    final record = contract.first;
    final customer = widget.store.customerFor(record.customerId);
    final space = widget.store.adSpaceFor(record.adSpaceId);
    setState(() {
      selectedIndex = 3;
      _contractFilter = ContractListFilter.all;
    });
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!mounted) return;
      showDialog<void>(
        context: context,
        builder: (context) => AlertDialog(
          title: Text(space?.name ?? 'Sözleşme'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text('Kiracı: ${customer?.name ?? 'Bilinmiyor'}'),
              Text(
                'Başlangıç: ${DateFormat('dd.MM.yyyy').format(record.startDate)}',
              ),
              Text('Bitiş: ${DateFormat('dd.MM.yyyy').format(record.endDate)}'),
              Text(
                'Aylık bedel: ${NumberFormat.currency(locale: 'tr_TR', symbol: '₺').format(record.monthlyPrice)}',
              ),
              if (record.isFinalized)
                const Text('Durum: Bitti olarak işaretlendi'),
            ],
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('Kapat'),
            ),
          ],
        ),
      );
    });
  }

  void openExpiringContracts() {
    if (!_canAccess('contracts')) return;
    setState(() {
      selectedIndex = 3;
      _contractFilter = ContractListFilter.upcoming;
    });
  }

  @override
  void dispose() {
    widget.store.removeListener(_onStoreChanged);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        final isDesktop = constraints.maxWidth >= 800;
        final visibleIndices = _visibleIndices.isEmpty ? [0] : _visibleIndices;
        final selectedVisibleIndex = visibleIndices.indexOf(selectedIndex);

        return Scaffold(
          body: SafeArea(
            child: Row(
              children: [
                if (isDesktop)
                  AppSidebar(
                    selectedIndex: selectedIndex,
                    visibleIndices: visibleIndices,
                    onDestinationSelected: _selectPage,
                  ),
                Expanded(
                  child: Column(
                    children: [
                      Container(
                        height: 72,
                        padding: EdgeInsets.symmetric(
                          horizontal: isDesktop ? 24 : 16,
                        ),
                        decoration: BoxDecoration(
                          color: Theme.of(context).colorScheme.surface,
                          border: Border(
                            bottom: BorderSide(
                              color: Theme.of(
                                context,
                              ).colorScheme.outlineVariant,
                            ),
                          ),
                        ),
                        child: Row(
                          children: [
                            Expanded(
                              child: Text(
                                pageTitles[selectedIndex],
                                maxLines: 1,
                                overflow: TextOverflow.ellipsis,
                                style: TextStyle(
                                  fontSize: 20,
                                  fontWeight: FontWeight.bold,
                                  color: Theme.of(
                                    context,
                                  ).colorScheme.onSurface,
                                ),
                              ),
                            ),
                            Icon(
                              Icons.account_circle_outlined,
                              color: Theme.of(
                                context,
                              ).colorScheme.onSurfaceVariant,
                              size: 28,
                            ),
                            if (isDesktop) ...[
                              const SizedBox(width: 8),
                              Text(
                                'Yönetici',
                                style: TextStyle(
                                  color: Theme.of(
                                    context,
                                  ).colorScheme.onSurface,
                                ),
                              ),
                            ],
                          ],
                        ),
                      ),
                      Expanded(child: getSelectedPage()),
                    ],
                  ),
                ),
              ],
            ),
          ),

          bottomNavigationBar: isDesktop
              ? null
              : NavigationBar(
                  selectedIndex: selectedVisibleIndex < 0
                      ? 0
                      : selectedVisibleIndex,
                  onDestinationSelected: (index) =>
                      _selectPage(visibleIndices[index]),
                  destinations: visibleIndices
                      .map((index) => AppSidebar.destinations[index])
                      .toList(),
                ),
        );
      },
    );
  }
}
