      final docs = results[0];
      final deletes = results[1];

      final deletedAt = <String, DateTime>{
        for (final doc in deletes.docs)
          if (doc.data()['collection'] == name &&
              doc.data()['recordId'] is String)
            doc.data()['recordId'] as String: _readTime(
              doc.data()['_syncUpdatedAt'],
            ),
      };

      final seen = <String>{};
      for (final doc in docs.docs) {
        final id = doc.id;
        seen.add(id);
        if (pushed.contains(id)) continue;
        final docTime = _readTime(doc.data()['_syncUpdatedAt']);
        final deletedTime = deletedAt[id];
        if (deletedTime != null && !deletedTime.isBefore(docTime)) {
          if (values.remove(id) != null) changed = true;
        } else {
          values[id] = _decode(name, {...doc.data(), 'id': id});
          changed = true;
        }
      }
      for (final entry in deletedAt.entries) {
        if (seen.contains(entry.key) || pushed.contains(entry.key)) continue;
        if (values.remove(entry.key) != null) changed = true;
      }
    }

    if (changed) {
      final list = values.values.toList();
      _assignList(name, list);
      await _saveList(name, list);
    }
    return changed;
  }

  Future<void> initialize() async {
    if (kIsWeb) return;
    final platform = defaultTargetPlatform;
    if (platform != TargetPlatform.android &&
        platform != TargetPlatform.iOS &&
        platform != TargetPlatform.windows) {
      return;
    }
    if (_ready) return;

    timezone_data.initializeTimeZones();
    timezone.setLocalLocation(timezone.getLocation('Europe/Istanbul'));
    await _plugin.initialize(
      settings: _initializationSettings(platform),
      onDidReceiveNotificationResponse: (response) =>
          _onNotificationResponse?.call(response),
    );
    final launchDetails = await _plugin.getNotificationAppLaunchDetails();
    final launchResponse = launchDetails?.notificationResponse;
    if (launchDetails?.didNotificationLaunchApp == true &&
        launchResponse != null) {
      _onNotificationResponse?.call(launchResponse);
    }
    if (platform == TargetPlatform.android) {
      await _plugin
          .resolvePlatformSpecificImplementation<
            AndroidFlutterLocalNotificationsPlugin
          >()
          ?.requestNotificationsPermission();
    }
    if (platform == TargetPlatform.iOS) {
      await _plugin
          .resolvePlatformSpecificImplementation<
            IOSFlutterLocalNotificationsPlugin
          >()
          ?.requestPermissions(alert: true, badge: true, sound: true);
    }
    _ready = true;

class _DatePartsField extends StatefulWidget {
  const _DatePartsField({
    required this.label,
    required this.initialDate,
    required this.onChanged,
    this.readOnly = false,
    this.validator,
  });

  final String label;
  final DateTime initialDate;
  final ValueChanged<DateTime> onChanged;
  final bool readOnly;
  final String? Function(DateTime)? validator;

  @override
  State<_DatePartsField> createState() => _DatePartsFieldState();
}

class _DatePartsFieldState extends State<_DatePartsField> {
  late final TextEditingController _day;
  late final TextEditingController _month;
  late final TextEditingController _year;

  @override
  void initState() {
    super.initState();
    _day = TextEditingController();
    _month = TextEditingController();
    _year = TextEditingController();
    _setText(widget.initialDate);
  }

  @override
  void didUpdateWidget(covariant _DatePartsField oldWidget) {
    super.didUpdateWidget(oldWidget);
    final newDate = ContractDateHelper.dateOnly(widget.initialDate);
    if (ContractDateHelper.dateOnly(oldWidget.initialDate) != newDate) {
      if (_readDate() == newDate) return;
      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (!mounted ||
            ContractDateHelper.dateOnly(widget.initialDate) != newDate) {
          return;
        }
        _setText(newDate);
      });
    }
  }

  @override
  void dispose() {
    _day.dispose();
    _month.dispose();
    _year.dispose();
    super.dispose();
  }

  void _setText(DateTime date) {
    _day.text = date.day.toString().padLeft(2, '0');
    _month.text = date.month.toString().padLeft(2, '0');
    _year.text = date.year.toString().padLeft(4, '0');
  }

  DateTime? _readDate() {
    final day = int.tryParse(_day.text);
    final month = int.tryParse(_month.text);
    final year = int.tryParse(_year.text);
    if (day == null ||
        month == null ||
        year == null ||
        year < 2000 ||
        year > 2100) {
      return null;
    }
    if (month < 1 || month > 12) return null;
    final lastDay = DateTime(year, month + 1, 0).day;
    if (day < 1 || day > lastDay) return null;
    return DateTime(year, month, day);
  }

  void _handleInput(FormFieldState<DateTime> fieldState) {
    final value = _readDate();
    fieldState.didChange(value);
    if (value != null) widget.onChanged(value);
  }

  Future<void> _pickDate(FormFieldState<DateTime> fieldState) async {
    final current = _readDate() ?? widget.initialDate;
    final picked = await showDatePicker(
      context: context,
      locale: const Locale('tr', 'TR'),
      initialDate: current,
      firstDate: DateTime(2000),
      lastDate: DateTime(2100, 12, 31),
    );
    if (picked == null || !mounted) return;
    _setText(picked);
    fieldState.didChange(picked);
    widget.onChanged(ContractDateHelper.dateOnly(picked));
  }

  Widget _partInput({
    required String hint,
    required TextEditingController controller,
    required int maxLength,
    required FormFieldState<DateTime> fieldState,
  }) => Expanded(
    flex: hint == 'YYYY' ? 3 : 2,
    child: TextFormField(
      controller: controller,
      textAlign: TextAlign.center,
      keyboardType: TextInputType.number,
      enabled: !widget.readOnly,
      inputFormatters: [
        FilteringTextInputFormatter.digitsOnly,
        LengthLimitingTextInputFormatter(maxLength),
      ],
      decoration: InputDecoration(
        hintText: hint,
        counterText: '',
        isDense: true,
        filled: true,
        contentPadding: const EdgeInsets.symmetric(horizontal: 6, vertical: 14),
        border: OutlineInputBorder(borderRadius: BorderRadius.circular(8)),
      ),
      onChanged: (_) => _handleInput(fieldState),
    ),
  );

  @override
  Widget build(BuildContext context) => FormField<DateTime>(
    initialValue: widget.initialDate,
    autovalidateMode: AutovalidateMode.onUserInteraction,
    validator: (_) {
      final date = _readDate();
      if (date == null) return 'GG / AA / YYYY biÃ§iminde geÃ§erli tarih girin.';
      return widget.validator?.call(date);
    },
    builder: (fieldState) => Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(widget.label, style: Theme.of(context).textTheme.labelLarge),
        const SizedBox(height: 8),
        Row(
          children: [
            _partInput(
              hint: 'GG',
              controller: _day,
              maxLength: 2,
              fieldState: fieldState,
            ),
            const SizedBox(width: 8),
            _partInput(
              hint: 'AA',
              controller: _month,
              maxLength: 2,
              fieldState: fieldState,
            ),
            const SizedBox(width: 8),
            _partInput(
              hint: 'YYYY',
              controller: _year,
              maxLength: 4,
              fieldState: fieldState,
            ),
            const SizedBox(width: 4),
            IconButton(
              tooltip: 'Takvimden seÃ§',
              onPressed: widget.readOnly ? null : () => _pickDate(fieldState),
              icon: const Icon(Icons.calendar_month_outlined),
            ),
          ],
        ),
        if (fieldState.errorText != null)
          Padding(

  ) async {
    final bytes = await buildAdSpacesPdf(store, adSpaces);

        'rst-reklam-alanlari-${DateTime.now().millisecondsSinceEpoch}.pdf';

    final file = File(
      '${Directory.systemTemp.path}${Platform.pathSeparator}$fileName',
    );

    await file.writeAsBytes(bytes, flush: true);

    await Process.run('cmd', ['/c', 'start', '', file.path], runInShell: true);
